// 스티키 헤더 / Sticky header
// 팩: lp (랜딩 페이지와 웹 효과) · element 03 · 분류: 스크롤 효과
// 동작 요약: 공지 띠는 스크롤되어 사라지지만 그 아래 헤더는 위쪽 경계에 멈추고 콘텐츠가 밑으로 지나갑니다.
// 동작 사양:
//   - 스크롤: 0.4–1.3 s에 0→110 px, 1.9–2.8 s에 110→250 px, 3.4–4.4 s에 250→0(cubic ease-in-out).
//   - 맨 위에 22 px 네이비 공지 띠, 아래에 34 px 헤더(로고·메뉴 세 개·버튼)를 둡니다.
//   - 헤더 위치 = max(22 − scroll, 0). 공지 띠가 사라지면 위쪽에 고정됩니다(position: sticky; top: 0).
//   - 그 다음 24 px 스크롤 동안 헤더 아래의 12 px 그림자 불투명도가 0에서 0.16으로 높아집니다.
//   - 사진·제목·카드 세 개의 콘텐츠는 헤더 아래에서 정상 스크롤됩니다.
// 활용 상황: 긴 페이지에서 메뉴와 주요 행동 버튼을 계속 사용할 수 있게 할 때 사용합니다.
// 태그: sticky, sticky-header, fixed-header, position-sticky, navigation, header, scroll
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://comnori.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#075
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 03
// 03 追従ヘッダー（スクロールすると、ヘッダーが画面の上で止まってついてくる）
CELLS[2] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.4, 0], [1.3, 110], [1.9, 110], [2.8, 250], [3.4, 250], [4.4, 0]], u);
  const STRIP = 22;
  view(k, () => {
    k.save(); k.translate(0, -y);
    pic(k, 24, 70, 214, 112, 0, 8);
    txt(k, '新しい毎日を、', 262, 104, `800 22px ${JP}`, C.ink); bars(k, 262, 120, 210, 3); btn(k, 262, 160, 100, 26, '詳しく見る');
    k.fillStyle = C.soft; k.fillRect(0, 212, VW, 240);
    txt(k, '特長', VW / 2, 246, `800 18px ${JP}`, C.ink, 'center');
    for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = '#fff'; rr(k, x, 262, 136, 120, 8); k.fill(); pic(k, x + 8, 270, 120, 58, i, 5); bars(k, x + 10, 340, 110, 3, C.faint, 6, 11); }
    k.fillStyle = C.ink; k.fillRect(0, 0, VW, STRIP);
    txt(k, '期間限定キャンペーン実施中', VW / 2, STRIP / 2 + 0.5, `600 11px ${JP}`, '#fff', 'center', 'middle');
    k.restore();
    const hy = Math.max(STRIP - y, 0), sh = clamp((y - STRIP) / 24);
    if (sh > 0) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.16 * sh})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    header(k, hy);
    sbar(k, y / 250, 0.42);
  });
};
