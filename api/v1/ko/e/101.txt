// 렘브란트 조명 / Rembrandt lighting
// 팩: photo (사진 표현·조명·스타일) · element 05 · 분류: 조명과 시간대
// 동작 요약: 옆 45°·위 45° 조명이 켜집니다. 얼굴 오른쪽은 어두워지지만 그쪽 뺨에 작은 역삼각형 빛이 남습니다.
// 동작 사양:
//   - 장면 1: y 150 수평선 앞의 상반신 인물(머리 중심 x 270, y 78, 반지름 29 px, 얼굴·머리카락 없음). 배경은 더 어두운 장면(하늘 #E7E5E1)으로 바뀝니다.
//   - 머리: x = 270 + 0.04·29 오른쪽은 그림자 #33405A, 왼쪽은 빛 #FFC75A입니다.
//   - 빛 삼각형: (x 270+0.24R, y 78−0.12R), (x 270+0.7R, y 78−0.12R), (x 270+0.24R, y 78+0.5R), R = 29(약 너비 13 px·높이 18 px). 너비 약 6 px의 그림자 띠가 밝은 쪽과 분리합니다.
//   - 몸: 중심 −0.45×56 px 왼쪽은 밝고 +0.3×56 px 오른쪽은 어둡습니다. 평면도에 조명 45°, 오른쪽 위에 “45° · ↑45°”를 표시합니다.
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 회화적이고 조각 같은 고전적인 인물 사진에 사용합니다.
// 태그: rembrandt lighting, portrait lighting, triangle, 45 degree light, chiaroscuro, painterly
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://comnori.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#101
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 05
// レンブラントライティング: 斜め45°・上からの光。影側の頬に三角形の光が残る
CELLS[4] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    kk.fillStyle = sh; kk.fillRect(PX + 0.04 * R, PY - R - 2, R * 2, R * 2 + 4);       // 鼻筋から右は影
    kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.clip();
    kk.fillStyle = lit; kk.beginPath();                                                   // 影側の頬に残る光の三角形
    kk.moveTo(PX + 0.24 * R, PY - 0.12 * R); kk.lineTo(PX + 0.7 * R, PY - 0.12 * R); kk.lineTo(PX + 0.24 * R, PY + 0.5 * R); kk.closePath(); kk.fill();
  });
  bust(k, PX, PY, R, 'rgba(0,0,0,0)', kk => {
    kk.fillStyle = 'rgba(0,0,0,0)';
    torso(kk, PX, PY, R, PX - W, PX - 0.45 * W, lit);
    torso(kk, PX, PY, R, PX + 0.3 * W, PX + W, sh);
  });
  plan(k, [[Math.PI / 4, a]]);
  hud(k, '45° · ↑45°');
});
