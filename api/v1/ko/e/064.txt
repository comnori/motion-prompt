// 페이지네이션 / Pagination
// 팩: ui (UI 컴포넌트 이름) · element 16 · 분류: 선택과 전환
// 동작 요약: 페이지 번호 2와 3을 누르면 파란 원이 옮겨가고 위 목록이 바뀝니다. 마지막에는 1로 돌아갑니다.
// 동작 사양:
//   - 542×191 카드에서 4.5 s 주기. “‹ 1 2 3 4 5 ›”를 y 160에 가운데 정렬합니다. 각 지름 36, 간격 8 px. 현재 페이지는 반지름 18의 파란 원과 흰 글자로 표시합니다.
//   - 커서는 1.0 s에 2, 2.1 s에 3, 3.3 s에 1을 클릭합니다.
//   - 파란 원은 0.2 s 동안 easeInOutCubic으로 페이지당 44 px 이동합니다(3에서 1로는 한 번에 88 px).
//   - 위의 세 줄 목록은 1.05, 2.15, 3.35 s 전후 각각 0.2 s 안에 불투명도 1 → 0 → 1로 변하며 최저점에서 내용이 교체됩니다.
// 활용 상황: 검색 결과·기사 등 긴 목록을 번호가 있는 페이지로 나눌 때 사용합니다.
// 태그: pagination, pager, page-numbers, paging, next-prev, navigation
// 크기: 542x191 px · 반복 주기: 4.5 s
// 필요한 코드: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#064
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 16
// 16 ページネーション（Pagination）
CELLS[15] = ui((w, h, t) => {
  const page = t < 1.05 ? 0 : t < 2.15 ? 1 : t < 3.35 ? 2 : 0;
  const fade = Math.min(...[1.05, 2.15, 3.35].map(c => Math.abs(t - c) < .2 ? Math.abs(t - c) / .2 : 1));
  X.save(); X.globalAlpha = fade;
  for (let k = 0; k < 3; k++) { const y = 10 + k * 40; fillRR(24, y, 52, 32, 6, (k + page) % 3 === 0 ? C.mb : C.line); bar(90, y + 4, [200, 160, 240, 180][(k + page) % 4], C.line2, 10); bar(90, y + 20, [320, 280, 360][(k + 2 * page) % 3], C.line, 7); }
  X.restore();
  const items = ['‹', '1', '2', '3', '4', '5', '›'], d = 36, g = 8, total = items.length * d + (items.length - 1) * g, x0 = (w - total) / 2;
  const pos = eio(seg(t, 1.0, 1.2)) + eio(seg(t, 2.1, 2.3)) - 2 * eio(seg(t, 3.3, 3.5));
  circle(x0 + (1 + pos) * (d + g) + d / 2, 160, d / 2, C.blue);
  items.forEach((s, i) => { const cx = x0 + i * (d + g) + d / 2; const on = i >= 1 && i <= 5 && Math.abs(pos - (i - 1)) < .5; text(s, cx, 161, i === 0 || i === 6 ? 22 : 16, on ? '#fff' : '#666', 700, 'center', NUMF); });
  const px = i => x0 + i * (d + g) + d / 2 + 4;
  pointer([[.3, w + 30, h + 30], [.95, px(2), 166], [1.6, px(2), 166], [2.05, px(3), 166], [2.8, px(3), 166], [3.25, px(1), 166], [3.65, px(1), 166], [4.25, w + 30, h + 30]], [1.0, 2.1, 3.3], t);
});
