// 팩: ui (UI 컴포넌트 이름) · element 22 · 분류: 탐색과 상태 표시
// 동작 요약: “전송”을 누르면 카드가 희게 덮이고 중앙의 파란 호가 회전합니다. 완료되면 체크로 바뀐 뒤 사라집니다.
// 동작 사양:
//   - 542×191 카드에서 4.5 s 주기. 커서는 1.0 s에 아래 중앙의 “전송” 버튼(120×40)을 클릭합니다.
//   - 1.05–1.2 s에 흰 덮개(불투명도 0.78)와 스피너가 카드 전체에 선형으로 나타납니다.
//   - 스피너 중심 (271, 92): 반지름 26·두께 6의 옅은 고리 위에 끝이 둥근 파란 243°(1.35π) 호를 놓고 초당 1.3회(468°/s) 회전합니다.
//   - 2.95–3.05 s에 스피너가 사라집니다. 2.95–3.2 s에 흰 체크가 있는 파란 원(반지름 28)을 배율 0.7 → 1로 표시합니다(easeOutCubic).
//   - 3.75–4.0 s에 전체가 선형으로 사라집니다.
// 활용 상황: 전송·저장처럼 소요 시간을 알 수 없는 짧은 대기를 표시할 때 사용합니다.
// 태그: spinner, loading, loader, activity-indicator, progress-circle, busy
// 크기: 542x191 px · 반복 주기: 4.5 s
// 필요한 코드: kit ui -> https://comnori.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#070
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 22
// 22 スピナー（Spinner）
CELLS[21] = ui((w, h, t) => {
  const load = seg(t, 1.05, 1.2) * (1 - seg(t, 2.95, 3.05));
  const done = seg(t, 2.95, 3.15) * (1 - seg(t, 3.75, 4.0));
  pageLines(24, 18, 494); strokeRR(24, 90, 494, 34, 7, C.line2, 1.5);
  fillRR(w / 2 - 60, 140, 120, 40, 8, C.blue); text('送信', w / 2, 160, 16, '#fff', 600, 'center');
  const ov = Math.max(load, done);
  if (ov > 0) {
    X.fillStyle = `rgba(255,255,255,${.78 * ov})`; X.fillRect(0, 0, w, h);
    if (load > 0) { X.save(); X.globalAlpha = load; ring(w / 2, 92, 26, '#e3e8f5', 6); X.beginPath(); const a0 = t * Math.PI * 2 * 1.3; X.arc(w / 2, 92, 26, a0, a0 + Math.PI * 1.35); X.strokeStyle = C.blue; X.lineWidth = 6; X.lineCap = 'round'; X.stroke(); X.restore(); }
    if (done > 0) { X.save(); X.globalAlpha = done; const s = .7 + .3 * eo(seg(t, 2.95, 3.2)); X.translate(w / 2, 92); X.scale(s, s); circle(0, 0, 28, C.blue); check(0, 0, 11, '#fff', 4); X.restore(); }
  }
  pointer([[.3, w + 30, h + 30], [.95, w / 2 + 6, 166], [1.5, w / 2 + 6, 166], [2.2, w + 30, h + 30]], [1.0], t);
});
