// 스크롤 페이드 인 / Fade-in on scroll
// 팩: lp (랜딩 페이지와 웹 효과) · element 02 · 분류: 스크롤 효과
// 동작 요약: 스크롤하면서 제목·카드·본문이 화면에 들어오면 각각 올라오며 나타납니다.
// 동작 사양:
//   - 스크롤: 0.35–1.0 s에 0→95 px, 1.5–2.1 s에 95→200 px, 3.2–4.4 s에 200→0(모두 cubic ease-in-out).
//   - 발동선은 뷰포트 높이의 85%(위에서 143 px)입니다. 요소의 위쪽 경계가 이를 넘어가면 발동합니다(Intersection Observer 방식).
//   - 발동 후 0.7 s 동안 불투명도 0→1, 아래쪽 오프셋 28 px → 0(cubic ease-out).
//   - 요소 세 개: 페이지 위에서 172 px의 제목, 206 px의 카드 세 개 행, 334 px의 본문. 처음에는 모두 화면 아래에 있고 카드 행은 함께 나타납니다.
//   - 위로 되돌아갈 때 요소가 화면 아래로 빠져나가면 보이지 않는 상태로 초기화하고 다음 진입 때 다시 재생합니다.
// 활용 상황: 긴 랜딩 페이지에서 스크롤에 따라 내용을 조금씩 등장시킬 때 사용합니다.
// 태그: fade-in, scroll-reveal, reveal-on-scroll, intersection-observer, aos, entrance, scroll
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://comnori.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#074
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 02
// 02 スクロールでフェードイン（画面に入った要素が、下から浮かび上がって現れる）
CELLS[1] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.0, 95], [1.5, 95], [2.1, 200], [3.2, 200], [4.4, 0]], t), y = SY(u);
  const LINE = VH * 0.85, DUR = 0.7, RISE = 28;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '暮らしを、軽く。', 34, 52, `800 24px ${JP}`, C.ink); bars(k, 34, 68, 190, 3); btn(k, 34, 108, 96, 26, '詳しく見る');
    pic(k, 286, 18, 196, 110, 0, 8);
    const items = [
      [172, () => txt(k, '選ばれる3つの理由', VW / 2, 172 + 20, `800 18px ${JP}`, C.ink, 'center')],
      [206, () => { for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = C.soft; rr(k, x, 206, 136, 106, 8); k.fill(); pic(k, x + 8, 214, 120, 56, i, 5); bars(k, x + 10, 280, 112, 2, '#CBD2DE'); } }],
      [334, () => bars(k, 110, 334, 290, 3, C.faint, 7, 13)]
    ];
    for (const [top, draw] of items) {
      const t0 = enteredAt(SY, top, LINE, u); if (t0 === null) continue;
      const q = eOut(P(u, t0, t0 + DUR));
      k.save(); k.globalAlpha = q; k.translate(0, RISE * (1 - q)); draw(); k.restore();
    }
    k.restore();
    sbar(k, y / 200, 0.5);
  });
};
