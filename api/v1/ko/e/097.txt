// 역광 / Backlighting
// 팩: photo (사진 표현·조명·스타일) · element 01 · 분류: 조명과 시간대
// 동작 요약: 해가 화면 밖에서 사람 머리 바로 뒤로 이동합니다. 인물 가장자리에 가는 빛이 생기고 화면에 따뜻한 안개가 끼며 카메라 쪽 면은 어두워집니다.
// 동작 사양:
//   - 장면 1: y 150 수평선 앞의 상반신 인물(머리 중심 x 270, y 78, 반지름 29 px, 얼굴·머리카락 없음). 밝은 면은 단색 호박색 #FFC75A, 그림자는 단색 #33405A입니다.
//   - 해: 반지름 20 px. 왼쪽 위 화면 밖(x 40, y −40)에서 머리 바로 뒤(x 270, y 56)로 이동합니다. 반지름의 2.6×와 1.7×인 단색 고리(#FFE3A3, 35%·60%)를 둡니다.
//   - 윤곽: 인물을 7 px 선으로 그린 뒤 안을 채워 바깥에 3.5 px의 밝은 가장자리를 남깁니다.
//   - 인물 #4D5A75 → #33405A, 하늘 #FAF9F7 → #FFF4DC. 화면 전체에 #FFE3A3을 22%로 덮습니다.
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 공기감 있고 빛나는 인물 사진이나 빛에 감싸인 듯한 이미지에 사용합니다.
// 태그: backlighting, backlit, contre-jour, sun behind, haze, glow, halo
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#097
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 01
// 逆光: 太陽が画面の外から頭の真後ろへ回り込み、輪郭が光って人物は暗くなる
CELLS[0] = cell((k, u) => {
  const a = env(u);
  const BACK = Object.assign({}, BASE, { sky: '#FFF4DC', sky2: '#FFEBC4', far: '#F6E2BE', ground: '#F7EAD3', bld: '#F7E9D2', winOff: '#EEDCBF', win: '#EEDCBF', tree: '#EEDCBF', edge: '#E3CFAE', post: '#CDB894', trunk: '#E3CFAE', bulb: '#E3CFAE' });
  stagePortrait(k, pal(BASE, BACK, a), { sun: [lerp(40, PX, a), lerp(-40, PY - 22, a), 20, 1] });
  // 輪郭の光: 胸像の形を太い線でなぞってから中を塗り、縁だけを残す
  k.save(); k.globalAlpha = a; bustPath(k, PX, PY, PR); k.strokeStyle = BASE.lit; k.lineWidth = 7; k.lineJoin = 'round'; k.stroke(); k.restore();
  bust(k, PX, PY, PR, mixc(BASE.person, BASE.shade, a));
  // かすみ: 全体に光の色を薄く重ねる
  k.save(); k.globalAlpha = 0.22 * a; k.fillStyle = BASE.glow; k.fillRect(0, 0, CW, CH); k.restore();
});
