// 스크롤 확대 / Zoom on scroll
// 팩: lp2 (랜딩 페이지와 웹 효과 2) · element 08 · 분류: 스크롤 효과
// 동작 요약: 스크롤하면 중앙의 작은 사진이 창 전체로 커지고 주변 글자는 사라지며 사진 위에 제목이 나타납니다.
// 동작 사양:
//   - 511 × 168 뷰포트. 스크롤 진행도 p: 0.3–2.3 s에 0→1, 3.0–4.4 s에 1→0(cubic ease-in-out). 한 번은 4.7 s(팩의 9.4 s 주기마다 두 번)입니다.
//   - 사진 틀: 중앙 155 × 90(x = 178, y = 58, 모서리 반지름 8)에서 전체 창 511 × 168(반지름 0)로 변합니다. 위치·너비·높이·반지름은 p에 비례합니다.
//   - 주변 제목과 글자는 p가 0→0.45일 때 사라집니다(불투명도 1 − 2.2p).
//   - p가 0.7→1일 때 사진 중앙에 흰 제목(26 px, 자간 2 px)이 나타납니다. 스크롤바 손잡이도 따라갑니다.
// 활용 상황: 여행·부동산·브랜드 사이트에서 방문자를 사진 속으로 끌어들이는 느낌에 사용합니다.
// 태그: zoom, scroll, image, scale, scroll-driven, hero, fullscreen
// 크기: 539x208 px · 반복 주기: 9.4 s
// 필요한 코드: kit lp2 -> https://comnori.github.io/motion-prompt/api/v1/kit/lp2.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#152
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 08
// 08 スクロールで拡大（スクロールに合わせて、小さな写真が窓いっぱいまで広がる）
CELLS[7] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const p = steps([[0.3, 0], [2.3, 1], [3.0, 1], [4.4, 0]], u);
  const x = lerp(178, 0, p), y = lerp(58, 0, p), ww = lerp(155, VW, p), hh = lerp(90, VH, p);
  view(k, () => {
    const a = 1 - clamp(p * 2.2);
    k.save(); k.globalAlpha = a;
    txt(k, '景色の中へ', VW / 2, 38, `800 19px ${JP}`, C.ink, 'center');
    bars(k, 34, 66, 120, 4, C.faint, 6, 14); bars(k, 357, 66, 120, 4, C.faint, 6, 14);
    k.restore();
    pic(k, x, y, ww, hh, 0, lerp(8, 0, p));
    const b = clamp((p - 0.7) / 0.3);
    if (b > 0) { k.save(); k.globalAlpha = b; txt(k, '絶景へようこそ', VW / 2, VH / 2 + 2, `800 26px ${JP}`, '#fff', 'center', 'middle', '2px'); k.restore(); }
    sbar(k, p * 0.6, 0.4);
  });
};
