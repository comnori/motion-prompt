// 팩: motion (모션 그래픽 움직임) · element 01 · 분류: 등장과 텍스트 연출
// 동작 요약: 앱 아이콘이 튀어 오르듯 나타난 뒤, 알림 배지가 조금 늦게 나타납니다.
// 동작 사양:
//   - 2.35 s마다 한 번 나타납니다(4.7 s 주기마다 두 번: 먼저 코랄색 바탕의 별, 다음은 파란 바탕의 하트).
//   - 아이콘: 116 px 둥근 정사각형(모서리 반지름 30). 0.28 s부터 스프링 1 − e^(−6.5t)·cos(15t)로 배율 0→1, 최대 약 1.26까지 커집니다.
//   - 동시에 스프링 1 − e^(−6t)·cos(11t)로 회전각 −0.35 rad(약 −20°) → 0을 적용합니다.
//   - 배지: 숫자 "1"이 있는 노란 원, 반지름 25, 아이콘 오른쪽 위(+52, −52). 0.62 s부터 스프링 1 − e^(−7t)·cos(17t)로 나타납니다.
//   - 퇴장: 1.86–2.12 s. 배율에 (1 − backIn, s = 2.2)를 곱하여 조금 부풀었다가 0으로 줄어듭니다.
//   - 바닥 그림자: 아이콘 배율에 따라 커지는 62 × 8 타원(배율은 0–1로 제한).
// 활용 상황: 아이콘·버튼·배지 같은 작은 요소를 가볍게 강조하며 등장시킬 때 사용합니다.
// 태그: pop, pop-in, spring, bounce, scale, icon, badge, entrance
// 크기: 539x208 px · 반복 주기: 4.7 s
// 필요한 코드: kit motion -> https://comnori.github.io/motion-prompt/api/v1/kit/motion.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#001
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 01
// 01 ポップイン（アプリのアイコンがポンと弾んで現れ、通知バッジが遅れてポン）
CELLS[0] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 2, n = Math.floor(u / T), lu = u - n * T, X = w / 2, Y = h / 2 + 4, S = 116;
  const col = n % 2 ? C.blue : C.coral;
  const out = backIn(P(lu, 1.86, 2.12), 2.2);
  const sc = spring(lu - 0.28, 15, 6.5) * (1 - out);
  const rot = (1 - spring(lu - 0.28, 11, 6)) * -0.35;
  // 床の影
  k.fillStyle = 'rgba(0,0,0,.28)'; k.beginPath(); k.ellipse(X, Y + 78, 62 * clamp(sc), 8 * clamp(sc), 0, 0, 7); k.fill();
  if (sc > 0.002) {
    k.save(); k.translate(X, Y); k.rotate(rot); k.scale(sc, sc);
    k.fillStyle = col; rr(k, -S / 2, -S / 2, S, S, 30); k.fill();
    k.fillStyle = 'rgba(255,255,255,.16)'; rr(k, -S / 2, -S / 2, S, S / 2, [30, 30, 0, 0]); k.fill();
    k.fillStyle = '#fff';
    if (n % 2) { k.save(); k.translate(0, 2); heartPath(k, 30); k.fill(); k.restore(); }
    else { star(k, 0, 2, 36, 0, 0.48); k.lineJoin = 'round'; k.strokeStyle = '#fff'; k.lineWidth = 6; k.stroke(); k.fill(); }
    k.restore();
    // 通知バッジ
    const bs = spring(lu - 0.62, 17, 7) * (1 - out);
    if (bs > 0.002) { k.save(); k.translate(X + 52 * sc, Y - 52 * sc); k.scale(bs, bs); k.fillStyle = C.yel; k.beginPath(); k.arc(0, 0, 25, 0, 7); k.fill(); k.strokeStyle = C.navy; k.lineWidth = 5; k.stroke(); txt(k, '1', 0, 2, `800 30px ${DISP}`, C.navy, 'center', 'middle'); k.restore(); }
  }
};
