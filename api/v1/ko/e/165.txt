// 팩: lp2 (랜딩 페이지와 웹 효과 2) · element 21 · 분류: 콘텐츠를 돋보이게 하는 효과
// 동작 요약: 파란 섹션과 흰 섹션의 경계에서 겹친 두 물결이 천천히 흔들립니다.
// 동작 사양:
//   - 511 × 168 뷰포트 위에는 파란 그라데이션 섹션(#3B64E8 → #8FA9F7), 아래에는 흰 섹션을 놓고 경계에 물결 두 개를 겹칩니다. 한 주기 4.7 s(9.4 s마다 두 번).
//   - 뒤 물결(흰색, 불투명도 0.45): 중심 y = 100, 파장 260 px·진폭 8 px와 파장 130 px·진폭 4 px인 사인파의 합입니다.
//   - 앞 물결(흰색): 중심 y = 110, 파장 300 px·진폭 9 px와 파장 110 px·진폭 3.5 px를 합하며 아래 흰 섹션으로 이어집니다.
//   - 위상 a = 2π·t/4.7일 때 뒤 물결은 a + 1과 −2a, 앞 물결은 −a와 2a + 0.5를 사용합니다. 각 두 사인파는 반대로 움직여 계속 변형되고 4.7 s마다 처음으로 돌아옵니다.
// 활용 상황: 바다·여름·어린이 관련 사이트 등에서 섹션 경계를 직선보다 부드럽게 표현할 때 사용합니다.
// 태그: shape-divider, wave, section-divider, svg, ambient, loop
// 크기: 539x208 px · 반복 주기: 9.4 s
// 필요한 코드: kit lp2 -> https://comnori.github.io/motion-prompt/api/v1/kit/lp2.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#165
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 21
// 21 シェイプディバイダー（セクションの境目の波の形が、ゆっくりゆれ続ける）
CELLS[20] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const a = 2 * Math.PI * u / CYC;
  const wave = (base, A1, L1, p1, A2, L2, p2) => { k.beginPath(); k.moveTo(0, VH + 1); for (let x = 0; x <= VW + 4; x += 4) k.lineTo(x, base + A1 * Math.sin(2 * Math.PI * x / L1 + p1) + A2 * Math.sin(2 * Math.PI * x / L2 + p2)); k.lineTo(VW, VH + 1); k.closePath(); };
  view(k, () => {
    const g = k.createLinearGradient(0, 0, 0, 120); g.addColorStop(0, '#3B64E8'); g.addColorStop(1, '#8FA9F7'); k.fillStyle = g; k.fillRect(0, 0, VW, VH);
    k.fillStyle = 'rgba(255,255,255,.9)'; k.beginPath(); k.arc(430, 44, 22, 0, 7); k.fill();
    txt(k, '夏のキャンペーン', 34, 42, `800 22px ${JP}`, '#fff'); btn(k, 34, 56, 96, 24, '詳しく見る', '#fff', C.blue, 11);
    k.fillStyle = 'rgba(255,255,255,.45)'; wave(100, 8, 260, a + 1, 4, 130, -2 * a); k.fill();
    k.fillStyle = '#fff'; wave(110, 9, 300, -a, 3.5, 110, 2 * a + 0.5); k.fill();
    txt(k, 'おすすめプラン', 34, 146, `800 15px ${JP}`, C.ink); bars(k, 170, 140, 290, 1, C.faint, 6);
    bars(k, 34, 156, 200, 1, C.faint, 5);
  });
};
