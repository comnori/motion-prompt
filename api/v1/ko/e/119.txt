// 동작 사양:
//   - 장면 3: 왼쪽에 흉상(머리 중심 x 140, y 82, 반지름 28 px), 오른쪽에 언덕·집·나무·해(x 432, y 54, 반지름 15 px)가 있습니다.
//   - 형태: 오른쪽 경계 바깥 30 px에 중심을 둔 긴 타원 네 개(주기 동안 ±6 px 흔들림), 세로 반지름은 화면 높이의 1.1×.
//   - 가로 반지름과 색: 250 px #FFD089 35%, 170 px #FF9A3D 40%, 95 px #FF5A3A 45%, 45 px #FF3B30 50%(각각 반지름 0에서 해당 값까지 커짐).
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 여름의 추억처럼 향수를 자아내고 우연히 찍힌 듯한 느낌에 사용합니다.

// 필요한 코드: kit photo -> https://comnori.github.io/motion-prompt/api/v1/kit/photo.js

// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 23
// ライトリーク: 右端から、フィルムに光がもれたような赤と橙のにじみが帯になって入る
CELLS[22] = cell((k, u) => {
  const a = env(u), w = Math.sin(u / CYC * TAU);
  stageLand(k, BASE);
  k.save();
  // 縦長の楕円を右端の外に中心を置いて重ねる（外ほど淡い黄、端ほど赤）
  const band = (rx, col, al) => { k.globalAlpha = al * a; k.fillStyle = col; k.beginPath(); k.ellipse(CW + 30 + 6 * w, CH / 2, rx * a, CH * 1.1, 0, 0, TAU); k.fill(); };
  band(250, '#FFD089', 0.35); band(170, '#FF9A3D', 0.4); band(95, '#FF5A3A', 0.45); band(45, '#FF3B30', 0.5);
  k.restore();
});
