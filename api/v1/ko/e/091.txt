// 무한 스크롤 / Infinite scroll
// 팩: lp (랜딩 페이지와 웹 효과) · element 19 · 분류: 로딩·탐색·배경 반복 효과
// 동작 요약: 목록 끝까지 스크롤하면 스피너가 나오고 새 항목 세 개가 추가되어 계속 읽을 수 있습니다.
// 동작 사양:
//   - 목록 항목은 331 × 44, 모서리 반지름 8, 간격 52이며 마지막 아래에 스피너를 둡니다(반지름 10, 0.8 s마다 한 바퀴).
//   - 스크롤: 0.25–0.85 s에 38→116 px, 1.15–1.75 s에 116→194 px(cubic ease-in-out). 목록 끝이 보이는 곳에서 멈춥니다.
//   - 2.6 s에 항목 세 개를 추가합니다. 각각 0.35 s 동안 불투명도 0→1로, 아래 10 px에서 올라옵니다(cubic ease-out). 스피너는 새 목록 끝으로 옮깁니다.
//   - 주기마다 세 항목(156 px)씩 전진합니다. 내용은 세 항목마다 반복되어 주기 끝과 시작이 같아 보입니다.
// 활용 상황: 페이지를 넘기지 않고 계속 읽는 피드나 뉴스 목록에 사용합니다.
// 태그: infinite-scroll, endless-scroll, load-more, feed, pagination, spinner, list
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#091
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 19
// 19 無限スクロール（いちばん下まで来ると読み込み表示が出て、続きが足される）
CELLS[18] = (k, w, h, u) => {
  frame(k);
  // 1周期で 3件（156px）ぶん進む。周期の終わりは始まりと同じ見た目（内容は3件ごとにくり返す）
  const PITCH = 52, BATCH = 3, L0 = 6, TL = 2.6;
  const y = steps([[0.25, 38], [0.85, 116], [1.15, 116], [1.75, 194], [4.8, 194]], u);
  const n = u < TL ? L0 : L0 + BATCH, end = 10 + n * PITCH;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.save(); k.translate(0, -y);
    for (let i = 0; i < n; i++) {
      const top = 10 + i * PITCH, v = i % 3, fresh = i >= L0 ? eOut(P(u, TL, TL + 0.35)) : 1;
      if (top - y > VH || top + 44 - y < 0) continue;
      k.save(); k.globalAlpha = fresh; k.translate(0, 10 * (1 - fresh));
      k.fillStyle = '#fff'; rr(k, 90, top, 331, 44, 8); k.fill(); pic(k, 98, top + 6, 48, 32, v, 5);
      bars(k, 158, top + 12, [180, 140, 200][v], 1, C.ink, 7); bars(k, 158, top + 26, [120, 150, 100][v], 1, C.faint, 6);
      k.restore();
    }
    spinner(k, VW / 2, end + 18, 10, u);
    k.restore();
  });
};
