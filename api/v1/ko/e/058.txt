// 팩: ui (UI 컴포넌트 이름) · element 10 · 분류: 선택과 전환
// 동작 요약: 오른쪽 화살표를 누를 때마다 세 이미지가 한 칸씩 왼쪽으로 이동하고 아래 표시점도 바뀝니다.
// 동작 사양:
//   - 542×191 카드에서 4.5 s 주기. 슬라이드는 330×138, 모서리 반지름 8, 간격 18 px로 중앙 배치하며 양옆 이미지가 조금 보입니다.
//   - 좌우에 화살표가 있는 흰 원형 버튼(반지름 18, 그림자 블러 10)을 둡니다. 아래 점 세 개 중 현재 것은 파란 16×8, 나머지는 8×8입니다.
//   - 커서는 오른쪽 화살표 (514, 88)를 1.0, 2.1, 3.2 s에 클릭합니다.
//   - 클릭마다 0.4 s 동안 easeInOutCubic으로 한 슬라이드와 간격을 합친 348 px만큼 왼쪽으로 이동합니다.
//   - 풍경·원 세 개·막대그래프 이미지가 반복되어 세 번 이동하면 처음 이미지로 돌아옵니다.
// 활용 상황: 한정된 공간에서 여러 이미지나 추천 항목을 하나씩 보여 줄 때 사용합니다.
// 태그: carousel, slider, slideshow, gallery, swipe, paging-dots
// 크기: 542x191 px · 반복 주기: 4.5 s
// 필요한 코드: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#058
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 10
// 10 カルーセル（Carousel）
CELLS[9] = ui((w, h, t) => {
  const p = eio(seg(t, 1.0, 1.4)) + eio(seg(t, 2.1, 2.5)) + eio(seg(t, 3.2, 3.6));
  const sw = 330, gap = 18, cx = w / 2;
  X.save(); X.beginPath(); X.rect(0, 0, w, 160); X.clip();
  for (let j = Math.floor(p) - 2; j <= Math.floor(p) + 2; j++) {
    const x = cx - sw / 2 + (j - p) * (sw + gap), idx = ((j % 3) + 3) % 3;
    if (idx === 0) mountains(x, 14, sw, 138);
    if (idx === 1) { fillRR(x, 14, sw, 138, 8, '#e3ebfd'); circle(x + 90, 83, 34, C.mb); circle(x + 165, 83, 34, C.sb); circle(x + 240, 83, 34, C.mb); }
    if (idx === 2) { fillRR(x, 14, sw, 138, 8, C.lb); for (let k = 0; k < 5; k++) { const bh = [40, 70, 55, 95, 80][k]; fillRR(x + 60 + k * 46, 132 - bh, 30, bh, 4, k === 3 ? C.blue : C.mb); } }
  }
  X.restore();
  [[30, 83, Math.PI / 2], [w - 30, 83, -Math.PI / 2]].forEach(([x, y, r]) => { shadowed(() => circle(x, y, 18, '#fff'), 10, 3, .2); chevron(x, y, 5, r, '#555'); });
  const act = Math.round(p) % 3;
  for (let k = 0; k < 3; k++) { const on = k === act; fillRR(w / 2 - 30 + k * 22 - (on ? 4 : 0), 172, on ? 16 : 8, 8, 4, on ? C.blue : C.line2); }
  pointer([[.3, w + 30, h + 30], [.95, w - 28, 88], [3.4, w - 28, 88], [4.2, w + 30, h + 30]], [1.0, 2.1, 3.2], t);
});
