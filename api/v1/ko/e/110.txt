// 어안 렌즈 / Fisheye lens
// 팩: photo (사진 표현·조명·스타일) · element 14 · 분류: 렌즈와 카메라 효과
// 동작 요약: 거리 중앙이 볼록해집니다. 모서리는 그대로이고 건물의 수직선과 도로의 선이 바깥으로 휘어집니다.
// 동작 사양:
//   - 장면 2: 거리 모퉁이. y 132–162의 도로 뒤에 건물·나무·가로등(x 114, 292, 494)이 있습니다. 왼쪽 앞에 사람이 서 있습니다(발 x 176, y 200, 키 124 px).
//   - 왜곡: 중심에서 거리 d인 점을 그 거리의 g = (1 + k) / (1 + k·d²/R²)배 위치로 옮깁니다. R은 중심에서 모서리까지 거리(289 px)입니다.
//   - k: 0 → 1.25. 중심은 2.25× 확대되고 모서리는 움직이지 않습니다.
//   - 왜곡 전 모든 윤곽선을 6 px마다 나누어 직선이 곡선이 되게 합니다.
//   - 오른쪽 위 초점거리: 24mm → 8mm.
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 스케이트보드·뮤직비디오 등에서 대담하고 장난스럽거나 구멍으로 들여다보는 듯한 시점에 사용합니다.
// 태그: fisheye, fisheye lens, barrel distortion, ultra wide, 8mm, distortion
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://comnori.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#110
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 14
// 魚眼レンズ: 中心がふくらみ、建物の縦の線が外側へ丸く曲がる
CELLS[13] = cell((k, u) => {
  const a = env(u), kk = 1.25 * a, cx = CW / 2, cy = CH / 2, R2 = cx * cx + cy * cy;
  const M = kk < 0.002 ? null : (x, y) => { const dx = x - cx, dy = y - cy, g = (1 + kk) / (1 + kk * (dx * dx + dy * dy) / R2); return [cx + dx * g, cy + dy * g]; };
  stageStreet(k, BASE, { M });
  hud(k, `${Math.round(lerp(24, 8, a))}mm`);
});
