// 그라디언트 애니메이션 / Animated gradient
// 팩: lp (랜딩 페이지와 웹 효과) · element 23 · 분류: 로딩·탐색·배경 반복 효과
// 동작 요약: 제목 뒤의 주황·연주황·파랑·연파랑 얼룩이 원을 그리며 떠다녀 색의 혼합이 계속 바뀝니다.
// 동작 사양:
//   - 기본색은 #FFF1EB입니다. 그 위에 방사형 그라데이션 네 개(중심색 → 가장자리 투명, 각각 불투명도 0.85)를 겹칩니다.
//   - 주황(반지름 230), 연주황 #FFB08F(210), 파랑(240), 연파랑 #9DB6FB(200).
//   - 각 중심은 4.8 s마다 타원을 한 바퀴 돕니다(x 반지름 70, 60, 80, 50; y 반지름 30, 40, 30, 35; 시작각 0, 2.1, 4.2, 1.0 rad).
//   - 위에 흰 제목(34 px)과 버튼을 놓습니다. 그라데이션만 움직입니다.
// 활용 상황: 사진 없이 히어로나 제목 뒤에 움직임과 색을 더할 때 사용합니다.
// 태그: gradient, animated-gradient, mesh-gradient, background, hero, ambient, loop
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://comnori.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#095
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 23
// 23 グラデーションアニメーション（背景のグラデーションの色が、ゆっくり流れるように動き続ける）
CELLS[22] = (k, w, h, u) => {
  frame(k);
  const a = 2 * Math.PI * u / CYC;                        // 4.8秒で1周
  const BLOBS = [[C.acc, 150, 60, 230, 70, 30, 0], ['#FFB08F', 380, 40, 210, 60, 40, 2.1], [C.blue, 330, 150, 240, 80, 30, 4.2], ['#9DB6FB', 90, 170, 200, 50, 35, 1.0]];
  view(k, () => {
    k.fillStyle = '#FFF1EB'; k.fillRect(0, 0, VW, VH);
    for (const [col, x0, y0, r, ax, ay, ph] of BLOBS) {
      const x = x0 + ax * Math.cos(a + ph), y = y0 + ay * Math.sin(a + ph);
      const g = k.createRadialGradient(x, y, 0, x, y, r); g.addColorStop(0, col); g.addColorStop(1, 'rgba(255,255,255,0)');
      k.globalAlpha = 0.85; k.fillStyle = g; k.fillRect(0, 0, VW, VH); k.globalAlpha = 1;
    }
    txt(k, 'Hello, Future.', 40, 82, `800 34px ${EN}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 40, 100, 200, 2, 'rgba(255,255,255,.8)', 6, 12, 0.6);
    btn(k, 40, 128, 104, 26, 'はじめる', '#fff', C.ink);
  });
};
