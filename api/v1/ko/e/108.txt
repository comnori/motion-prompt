// 패닝 샷 / Panning shot
// 팩: photo (사진 표현·조명·스타일) · element 12 · 분류: 렌즈와 카메라 효과
// 동작 요약: 카메라가 자동차를 따라가 차는 중앙에 선명하게 남고 뒤의 건물·나무·가로등만 옆으로 길게 흐려집니다.
// 동작 사양:
//   - 거리 모퉁이: y 132–162의 도로 뒤에 건물·나무·가로등이 있습니다(앞의 큰 사람 없음).
//   - 자동차: 효과가 없으면 5 s 동안 x −80에서 620으로 이동하고, 효과가 있으면 x 270 ±10 px에 머뭅니다.
//   - 잔상: 배경을 가로 46 px에 걸쳐 12번 그린 뒤 평균합니다(퍼짐 0 → 46 px).
//   - 바퀴: 스포크 선이 5 s마다 12번 회전합니다.
//   - 오른쪽 위 셔터 속도: 1/1000 → 1/15.
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 정지 사진에서 자동차·자전거·달리는 사람의 속도를 표현할 때 사용합니다.
// 태그: panning, panning shot, motion blur, speed, sports photography, slow shutter
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#108
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 12
// 流し撮り: 走る車を追ってカメラを振ると、車はくっきり、背景だけ横に流れる
CELLS[11] = cell((k, u) => {
  const a = env(u);
  const drive = lerp(-80, 620, u / CYC), x = lerp(drive, 270 + 10 * Math.sin(u / CYC * TAU), a);
  const S = 46 * a, n = S < 0.5 ? 1 : 12;
  for (let i = 0; i < n; i++) {   // 背景を横にずらして重ね、平均をとる（流れ）
    k.save(); k.globalAlpha = 1 / (i + 1); k.translate(n > 1 ? lerp(-S / 2, S / 2, i / (n - 1)) : 0, 0); stageStreet(k, BASE, { car: false, person: false }); k.restore();
  }
  const D = pen(k); car(D, x, BASE, { head: BASE.lit });
  // 車輪の回転（1周に12回転＝継ぎ目なし）
  k.save(); k.strokeStyle = BASE.tire; k.lineWidth = 1.6;
  for (const wx of [x - 21, x + 21]) { const t = u / CYC * TAU * 12; k.beginPath(); k.moveTo(wx - Math.cos(t) * 5, 156 - Math.sin(t) * 5); k.lineTo(wx + Math.cos(t) * 5, 156 + Math.sin(t) * 5); k.stroke(); }
  k.restore();
  hud(k, a > 0.5 ? '1/15' : '1/1000');
});
