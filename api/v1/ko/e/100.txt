// 스플릿 조명 / Split lighting
// 팩: photo (사진 표현·조명·스타일) · element 04 · 분류: 조명과 시간대
// 동작 요약: 조명이 정면 가까운 20°에서 바로 옆 90°로 이동합니다. 머리의 그림자가 초승달 모양에서 정확한 절반까지 커져 얼굴을 양분합니다.
// 동작 사양:
//   - 장면 1: y 150 수평선 앞의 상반신 인물(머리 중심 x 270, y 78, 반지름 29 px, 얼굴·머리카락 없음). 배경은 더 어두운 장면(하늘 #E7E5E1)으로 바뀝니다.
//   - 조명: 화면 왼쪽에서 각도 20° → 90°(오른쪽 아래 평면도, 오른쪽 위 “90° · ↑0°” 표시).
//   - 머리: x = 270 + cos(angle)·√(29² − dy²) 오른쪽은 그림자 #33405A, 왼쪽은 빛 #FFC75A입니다. 90°에서는 경계가 곧은 중앙선입니다.
//   - 몸(반너비 56 px): x = 270 + cos(angle)·56 오른쪽은 그림자이고 왼쪽 가장자리의 띠는 밝습니다.
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 극적이거나 긴장감 있고 분위기 있는 인물 사진에 사용합니다.
// 태그: split lighting, side light, portrait lighting, half face, dramatic, chiaroscuro
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://comnori.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#100
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 04
// スプリットライティング: 光が真横（90°）へ回り、顔の右半分が影に沈む
CELLS[3] = cell((k, u) => {
  const a = env(u), az = lerp(0.35, Math.PI / 2, a);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a), W = torsoW(PR);
  bust(k, PX, PY, PR, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, PR, 0, TAU); kk.fill();
    headShade(kk, PX, PY, PR, az, 1, sh);
    torso(kk, PX, PY, PR, PX - W, PX - Math.cos(az) * 0.4 * W - 0.4 * W, lit);
    torso(kk, PX, PY, PR, PX + Math.cos(az) * W, PX + W, sh);
  });
  plan(k, [[az, a]]);
  hud(k, `${Math.round(az * 180 / Math.PI)}° · ↑0°`);
});
