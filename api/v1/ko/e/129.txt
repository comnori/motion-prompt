// 텍스트 스크램블 / Text scramble
// 팩: motion2 (모션 그래픽 움직임 2) · element 09 · 분류: 텍스트와 도형
// 동작 요약: 무작위 문자가 계속 바뀌다가 왼쪽부터 올바른 단어로 확정됩니다.
// 동작 사양:
//   - 처음 2.35 s는 HELLO, 다음은 WORLD. 고정폭 글자 다섯 개(92 px)입니다.
//   - 미확정 칸에는 A–Z(I·O 제외), 0–9, #$%&@?의 문자를 무작위로 표시합니다. 4.7 s당 76번(약 0.062 s마다) 바뀌며 민트색·불투명도 75%입니다.
//   - 왼쪽부터 확정합니다: 칸 i는 0.45 + 0.18i s(마지막 1.17 s). 확정 직후 0.22 s는 노랑, 이후 흰색이며 배율 1 + 0.2·e^(−14t)로 안정됩니다.
//   - 1.85 s부터 오른쪽에서 왼쪽 순으로 무작위 문자로 돌아갑니다(칸 i는 1.85 + 0.08 × (4 − i) s).
//   - 각 칸 아래 막대는 확정 시 파랑, 다음 확정 대상은 노랑, 나머지는 네이비입니다.
// 활용 상황: 제목·숫자·코드명을 해독·해커·기술적인 느낌으로 등장시킬 때 사용합니다.
// 태그: text-scramble, scramble, decode, shuffle, random, text, hacker
// 크기: 539x208 px · 반복 주기: 4.7 s
// 필요한 코드: kit motion2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion2.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#129
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 09
// 09 テキストスクランブル（ランダムな文字が入れ替わり続け、左から順に正しい単語に決まる）
const SCR_GL = 'ABCDEFGHJKLMNPQRSTUVWXYZ0123456789#$%&@?';
CELLS[8] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const T = CYC / 2, n = Math.floor(u / T), lu = u - n * T, word = n ? 'WORLD' : 'HELLO', N = 5;
  const f = `700 92px Menlo, "SF Mono", monospace`; k.font = f;
  const cw = k.measureText('M').width + 16, x0 = w / 2 - cw * (N - 1) / 2, Y = h / 2 - 8;
  const step = Math.floor(u / (CYC / 76));
  for (let i = 0; i < N; i++) {
    const tl = 0.45 + i * 0.18, tu = 1.85 + (N - 1 - i) * 0.08, locked = lu >= tl && lu < tu;
    let ch, col, sc = 1;
    if (locked) { ch = word[i]; const age = lu - tl; col = age < 0.22 ? C.yel : '#fff'; sc = 1 + 0.2 * Math.exp(-age * 14); }
    else { ch = SCR_GL[Math.floor(H01(step * 131 + i * 977 + n * 31) * SCR_GL.length)]; col = C.mint; }
    k.save(); k.translate(x0 + i * cw, Y); k.scale(sc, sc); k.globalAlpha = locked ? 1 : 0.75; txt(k, ch, 0, 0, f, col, 'center', 'middle'); k.restore();
    const next = !locked && lu < tl && (i === 0 || lu >= 0.45 + (i - 1) * 0.18);
    k.fillStyle = locked ? C.blue : next ? C.yel : C.navy3; rr(k, x0 + i * cw - cw / 2 + 8, Y + 58, cw - 16, 8, 4); k.fill();
  }
};
