// 가변 글꼴 애니메이션 / Variable font animation
// 팩: motion2 (모션 그래픽 움직임 2) · element 12 · 분류: 텍스트와 도형
// 동작 요약: 같은 단어가 가는 글꼴에서 아주 굵은 글꼴로 바뀌었다 돌아오며 아래 슬라이더가 굵기를 보여 줍니다.
// 동작 사양:
//   - MOTION(112 px, system-ui, 자간 2 px). 굵기는 4.7 s마다 두 번 왕복합니다: v = 0.5 − 0.5·cos(2π × 2 × time ÷ 4.7), 양끝에서 감속합니다.
//   - 굵기 = round(100 + 800v), 100부터 900까지 font weight에 직접 적용합니다.
//   - 아래 슬라이더(x = 112→427, y = 180)의 왼쪽은 가늘게, 오른쪽은 굵게입니다. 노란 손잡이(반지름 14)가 v를 따라가며 위에 현재 굵기를 표시합니다.
// 활용 상황: 제목·로고·단어 하나의 글자 자체로 소리의 크기나 감정을 표현할 때 사용합니다.
// 태그: variable-font, font-weight, typography, text, weight, kinetic-type
// 크기: 539x208 px · 반복 주기: 4.7 s
// 필요한 코드: kit motion2 -> https://comnori.github.io/motion-prompt/api/v1/kit/motion2.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#132
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 12
// 12 可変フォントアニメーション（同じ単語の太さが、細→極太→細となめらかに変わる）
CELLS[11] = (k, w, h, u) => {
  stage(k);
  const v = 0.5 - 0.5 * Math.cos(2 * Math.PI * 2 * u / CYC), wt = Math.round(100 + 800 * v);
  txt(k, 'MOTION', w / 2, 80, `${wt} 112px system-ui`, '#fff', 'center', 'middle', '2px');
  const x0 = 112, x1 = w - 112, y = 180, kx = x0 + (x1 - x0) * v;
  txt(k, '細', x0 - 34, y + 1, `300 30px ${JP}`, C.dim, 'center', 'middle');
  txt(k, '太', x1 + 34, y + 1, `900 30px ${JP}`, C.dim, 'center', 'middle');
  k.fillStyle = C.navy3; rr(k, x0, y - 5, x1 - x0, 10, 5); k.fill();
  k.fillStyle = C.blue; rr(k, x0, y - 5, Math.max(10, kx - x0), 10, 5); k.fill();
  k.fillStyle = C.yel; k.beginPath(); k.arc(kx, y, 14, 0, 7); k.fill();
  txt(k, String(wt), kx, y - 30, `800 24px ${DISP}`, C.yel, 'center', 'middle');
};
