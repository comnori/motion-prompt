// 패럴랙스 스크롤 / Parallax scrolling
// 팩: lp (랜딩 페이지와 웹 효과) · element 01 · 분류: 스크롤 효과
// 동작 요약: 페이지를 아래로 스크롤하면 제목은 보통 속도로 올라가고 하늘·해·먼 언덕·가까운 언덕은 더 느리게 움직입니다.
// 동작 사양:
//   - 뷰포트가 511 × 168인 브라우저 창입니다. 스크롤: 0.3–1.9 s에 0→120 px, 2.9–4.3 s에 120→0(cubic ease-in-out).
//   - 각 레이어의 이동량은 스크롤 × 계수입니다: 하늘과 해 0.15, 먼 언덕 0.4, 가까운 언덕 0.7.
//   - 제목(28 px)·버튼·다음 섹션은 1.0의 일반 스크롤 속도로 움직입니다.
//   - 다음 흰 섹션은 히어로 아래(168 px)에서 시작해 느린 배경 위를 덮으며 올라옵니다.
//   - 오른쪽 스크롤바 손잡이는 스크롤에 맞춰 움직입니다.
// 활용 상황: 히어로나 섹션 사진에 깊이감을 줄 때 사용합니다.
// 태그: parallax, parallax-scrolling, scroll, depth, layers, hero, background
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#073
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 01
// 01 パララックススクロール（スクロールすると、空・山・見出しが違う速さで動く）
CELLS[0] = (k, w, h, u) => {
  frame(k);
  const s = steps([[0.3, 0], [1.9, 1], [2.9, 1], [4.3, 0]], u), y = s * 120;
  view(k, () => {
    const g = k.createLinearGradient(0, 0, 0, 200); g.addColorStop(0, '#9DB6FB'); g.addColorStop(1, '#E6ECFF');
    k.save(); k.translate(0, -y * 0.15);                         // 空と太陽: 0.15倍
    k.fillStyle = g; k.fillRect(0, 0, VW, 260); k.fillStyle = C.acc; k.beginPath(); k.arc(392, 54, 25, 0, 7); k.fill(); k.restore();
    k.save(); k.translate(0, -y * 0.4); mount(k, VW, 190, M1, '#7D9AF4', 0); k.restore();   // 遠い山: 0.4倍
    k.save(); k.translate(0, -y * 0.7); mount(k, VW, 205, M2, '#4B6BDB', 1); k.restore();   // 近い山: 0.7倍
    k.save(); k.translate(0, -y);                                // 見出し・ボタン・次のセクション: 1倍
    txt(k, '山へ、行こう。', 34, 64, `800 28px ${JP}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 34, 78, 160, 2, 'rgba(255,255,255,.85)');
    btn(k, 34, 106, 92, 26, '予約する');
    k.fillStyle = '#fff'; k.fillRect(0, VH, VW, 200);
    txt(k, '人気のコース', 34, VH + 34, `800 16px ${JP}`, C.ink);
    for (let i = 0; i < 3; i++) { pic(k, 34 + i * 152, VH + 50, 136, 58, i, 6); bars(k, 34 + i * 152, VH + 116, 110, 1, C.faint); }
    k.restore();
    sbar(k, s * 0.35, 0.4);
  });
};
