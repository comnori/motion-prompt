// 커서 팔로워 / Cursor follower
// 팩: lp (랜딩 페이지와 웹 효과) · element 11 · 분류: 호버와 클릭 상호작용
// 동작 요약: 커서가 페이지를 움직이면 주황색 원이 조금 뒤처져 따라옵니다.
// 동작 사양:
//   - 커서 경로: x = 255 + 175·sin(2πt/4.8), y = 86 + 46·sin(4πt/4.8). 옆으로 누운 팔자 경로를 4.8 s마다 한 바퀴 돕니다.
//   - 원: 반지름 17, 주황색, 불투명도 0.88. 커서 아래에 그립니다.
//   - 추적 규칙: 매 프레임(60 fps) 남은 거리의 12%를 이동합니다(x += (cursor x − x) × 0.12).
//   - 화살표 커서는 계속 표시합니다.
// 활용 상황: 브랜드 사이트나 포트폴리오에 직접 반응하는 듯한 재미를 더할 때 사용합니다.
// 태그: cursor, custom-cursor, cursor-follower, mouse-follow, lerp, trail, interaction
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://comnori.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#083
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 11
// 11 マウスストーカー（カーソルのあとを、丸が少し遅れて追いかける）
CELLS[10] = (k, w, h, u) => {
  frame(k);
  const cur = t => [255 + 175 * Math.sin(2 * Math.PI * t / CYC), 86 + 46 * Math.sin(4 * Math.PI * t / CYC)];
  // 毎フレーム（60fps）残りの距離の 12% ずつ近づく
  let t = u - 2, [fx, fy] = cur(t);
  for (let i = 0; i < 120; i++) { t += 1 / 60; const [tx, ty] = cur(t); fx += (tx - fx) * 0.12; fy += (ty - fy) * 0.12; }
  view(k, () => {
    txt(k, 'CREATIVE STUDIO', VW / 2, 74, `800 34px ${EN}`, C.ink, 'center', 'alphabetic', '3px');
    bars(k, VW / 2 - 120, 96, 240, 2, C.faint, 6, 12, 0.7);
    btn(k, VW / 2 - 50, 128, 100, 24, '作品を見る', C.ink);
    k.fillStyle = C.acc; k.globalAlpha = 0.88; k.beginPath(); k.arc(fx, fy, 17, 0, 7); k.fill(); k.globalAlpha = 1;
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
