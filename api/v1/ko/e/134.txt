// 동작 요약: 한 도형을 원형으로 복제하고 복사본들이 조금씩 다른 시점에 회전하며 맥동합니다.
// 동작 사양:
//   - 둥근 막대(24 × 60, 모서리 반지름 12)를 중앙의 반지름 58 원 주위에 최대 12번 복제하며 각각 바깥쪽을 향하게 합니다(각도 + 90°).
//   - 개수: 0.35–1.7 s에 1→12, 3.6–4.45 s에 12→1(quadratic ease-in-out). 새 복사본은 이전 자리에서 30° 더 떨어진 자기 자리로 이동합니다(cubic ease-out).
//   - 전체 고리는 4.7 s마다 한 바퀴 돕니다. 각 복사본 크기는 4.7 s마다 두 번 1 ± 0.22로 맥동하며 이전 것보다 0.55 rad 늦어 원형 파동을 만듭니다.
//   - 색은 파랑·코랄·노랑·민트 순입니다. 원본에는 흰 윤곽선이 있고 오른쪽 위에 ×N 카운터를 표시합니다.
// 활용 상황: 한 도형을 반복하는 로더·배경 무늬·꽃·광선 등에 사용합니다.
// 태그: repeater, duplicate, radial, array, clone, pattern, loader
// 크기: 539x208 px · 반복 주기: 4.7 s
// 필요한 코드: kit motion2 -> https://comnori.github.io/motion-prompt/api/v1/kit/motion2.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#134
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 14
// 14 リピーター（1つの図形が複製されて円に並び、少しずつずれながら回る）
CELLS[13] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const N = 12, X = w / 2, Y = h / 2, R = 58, spin = 2 * Math.PI * u / CYC;
  const cnt = 1 + (N - 1) * (eIOs(P(u, 0.35, 1.7)) - eIOs(P(u, 3.6, 4.45)));
  for (let i = N - 1; i >= 0; i--) {
    const vis = i === 0 ? 1 : clamp(cnt - i); if (vis <= 0) continue;
    const ang = spin + (i === 0 ? 0 : (i - 1 + eOut(vis))) * 2 * Math.PI / N;
    const pul = (1 + 0.22 * Math.sin(2 * Math.PI * 2 * u / CYC - i * 0.55)) * (0.55 + 0.45 * vis);
    k.save(); k.translate(X + Math.cos(ang) * R, Y + Math.sin(ang) * R); k.rotate(ang + Math.PI / 2); k.scale(pul, pul);
    k.fillStyle = PAL[i % 4]; rr(k, -12, -30, 24, 60, 12); k.fill();
    if (i === 0) { k.strokeStyle = '#fff'; k.lineWidth = 4; rr(k, -12, -30, 24, 60, 12); k.stroke(); }
    k.restore();
  }
  txt(k, '×' + Math.round(cnt), w - 34, 46, `800 40px ${DISP}`, C.yel, 'right', 'middle');
};
