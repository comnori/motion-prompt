// 팩: photo (사진 표현·조명·스타일) · element 08 · 분류: 조명과 시간대
// 동작 요약: 해가 지면 하늘이 두 층의 짙은 파랑으로 바뀌고 창문과 가로등에는 호박색 불이 켜지며 인물은 어두운 파랑이 됩니다.
// 동작 사양:
//   - 장면 1: y 150 수평선 앞의 상반신 인물(머리 중심 x 270, y 78, 반지름 29 px, 얼굴·머리카락 없음).
//   - 해: 반지름 18 px, x 120. 효과가 70%에 도달할 때까지 y 40에서 y 190으로 내려가고 40%–80% 구간에 사라집니다.
//   - 색: 하늘 위 #2C3B5E, 수평선 부근 #4A6191, 지면 #283452, 건물 #34446B, 인물 #1C2439.
//   - 조명: 창문 세 개 중 하나와 가로등 전구가 #FFC75A가 됩니다. 전구 주위에 13 px·8 px 고리를 둡니다(#FFE3A3, 35%·60%).
//   - 오른쪽 위 시간: 12:00 → 18:35.
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 푸른 하늘과 빛나는 조명을 함께 담는 도시·야경에 사용합니다.
// 태그: blue hour, twilight, dusk, night city, cityscape, deep blue sky
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#104
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 08
// ブルーアワー: 日が沈んだ後、空が深い青になり、窓と街灯の暖かい光が灯る
CELLS[7] = cell((k, u) => {
  const a = env(u);
  const p = pal(BASE, NIGHT, a);
  stagePortrait(k, p, {
    sun: [120, lerp(40, 190, P(a, 0, 0.7)), 18, 1 - P(a, 0.4, 0.8)],
    after: D => { const k2 = D.k; k2.save(); k2.globalAlpha = 0.35 * a; D.circ(413, HZ - 53, 13, BASE.glow); k2.globalAlpha = 0.6 * a; D.circ(413, HZ - 53, 8, BASE.glow); k2.restore(); D.circ(413, HZ - 53, 4, p.bulb); }
  });
  bust(k, PX, PY, PR, p.person);
  hud(k, clock(lerp(720, 1115, a)));
});
