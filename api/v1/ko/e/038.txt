// 스피드 램프 / Speed ramp
// 팩: camera (카메라 움직임과 장면 전환) · element 14 · 분류: 촬영 방식과 시간 연출
// 동작 요약: 한 쇼트 안에서 재생 속도가 빠름에서 느림으로 부드럽게 바뀌며 왼쪽 아래 그래프가 속도를 보여 줍니다.
// 동작 사양:
//   - 카메라: 완전히 고정(높이 1.3 m). 재생 속도만 바뀝니다.
//   - 재생 속도: 2.4× → 0.25×. 0.7 s 동안(1.2 s부터 1.9 s) quadratic ease-in-out(약 cubic-bezier(0.45,0,0.55,1))으로 바뀝니다.
//   - 피사체: 8 m 떨어진 사람이 왼쪽 6 m에서 오른쪽으로 달립니다. x = -6 + 영상 시간 × 2.0 m. 빠른 구간에서는 급하게 다가옵니다.
//   - 점프: x가 -1.5 m에서 2.5 m 사이일 때 최고점 0.99 m의 포물선을 그리며 양팔을 올립니다. 약 0.94 s에 도약해 대부분의 공중 동작을 슬로 모션으로 보여 주고 약 3.68 s에 착지합니다.
//   - 그래프: 왼쪽 아래의 120×40 px 속도 곡선에서 파란 점(반지름 5 px)이 현재 시간을 표시합니다.
//   - 복귀: 4.2 s에 멈추고 마지막 0.5 s 동안 크로스페이드합니다. 오른쪽 위에 속도를 표시합니다(×2.4 → ×0.3).
// 활용 상황: 도입부는 빠르게 유지하면서 동작의 핵심 부분만 느리게 보여 줄 때 사용합니다.
// 태그: speed ramp, time remap, ramping, fast to slow, variable speed, velocity
// 크기: 539x208 px · 반복 주기: 9.4 s
// 필요한 코드: kit camera -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#038
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 14
// 14 スピードランプ: 速い→遅い（グラフで速さを表示）
function rampTime(u) {   // 再生速度 v(u): 2.4 → 0.25（u 1.2〜1.9 で切替）
  const v = x => lerp(2.4, 0.25, eIOs(P(x, 1.2, 1.9)));
  let s = 0; const N = 60; for (let i = 0; i < N; i++) { const x = u * (i + .5) / N; s += v(x) * u / N; } return { s, v: v(u) };
}
CELLS[13] = shot((k, w, h, u) => {
  const uu = Math.min(u, CYC - RESET);
  const draw = (kk, uu) => {
    const { s, v } = rampTime(uu);
    const x = -6 + s * 2.0, q = P(x, -1.5, 2.5);
    world(kk, w, h, cam({ y: 1.3 }), {
      person: { x, z: 8, walk: s * 9, y: jumpY(q) * 0.9, armUp: q > 0 && q < 1 },
      after: (k2) => {   // 速度グラフ
        const gx = -w / 2 + 26, gy = h / 2 - 22, gw = 120, gh = 40;
        k2.fillStyle = 'rgba(255,255,255,.85)'; rr(k2, gx - 8, gy - gh - 10, gw + 16, gh + 18, 6); k2.fill();
        k2.strokeStyle = C.faint; k2.lineWidth = 1.5; k2.beginPath(); k2.moveTo(gx, gy); k2.lineTo(gx + gw, gy); k2.stroke();
        k2.strokeStyle = C.blue; k2.lineWidth = 2.5; k2.beginPath();
        for (let i = 0; i <= 40; i++) { const xx = i / 40 * (CYC - RESET), vv = rampTime(xx).v; const X = gx + i / 40 * gw, Y = gy - vv / 2.4 * gh; i ? k2.lineTo(X, Y) : k2.moveTo(X, Y); }
        k2.stroke();
        const X = gx + uu / (CYC - RESET) * gw, Y = gy - v / 2.4 * gh; k2.fillStyle = C.blue; k2.beginPath(); k2.arc(X, Y, 5, 0, 7); k2.fill();
      }
    });
    return v;
  };
  const r = P(u, CYC - RESET, CYC); let v;
  blend(k, kk => { v = draw(kk, uu); }, kk => draw(kk, 0), eIOs(r));
  return `×${(r > .5 ? 2.4 : v).toFixed(1)}`;
});
