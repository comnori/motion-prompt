// 지연 로딩 / Lazy loading
// 팩: lp (랜딩 페이지와 웹 효과) · element 18 · 분류: 로딩·탐색·배경 반복 효과
// 동작 요약: 회색 자리 표시자가 스크롤되어 들어오면 각각 스피너가 돌고 이미지가 도착하며 나타납니다. 첫 행은 처음부터 로드되어 있습니다.
// 동작 사양:
//   - 151 × 92 자리 표시자의 3 × 3 격자이며 첫 행은 처음부터 표시합니다. 스크롤은 0.35–1.1 s에 0→110 px, 1.7–2.5 s에 110→225 px, 3.5–4.5 s에 225→0입니다(cubic ease-in-out).
//   - 자리 표시자의 위쪽이 뷰포트 하단(168 px)을 넘으면 로딩을 시작하며 각 열은 왼쪽 열보다 0.12 s 늦게 시작합니다.
//   - 0.45 s 동안 자리 표시자 중앙에서 스피너가 회전합니다(반지름 11, 0.8 s마다 한 바퀴).
//   - 이후 0.35 s 동안 불투명도 0→1로 이미지가 나타납니다(cubic ease-out). 아직 로드하지 않은 곳에는 이미지 아이콘을 표시합니다.
//   - 화면 아래로 나간 자리 표시자는 보이지 않는 상태로 초기화합니다.
// 활용 상황: 이미지가 많은 페이지의 첫 화면 로딩을 빠르게 할 때 사용합니다.
// 태그: lazy-loading, lazy-load, images, placeholder, performance, spinner, gallery
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://comnori.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#090
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 18
// 18 遅延読み込み（画面に入った画像だけ、その時点で読み込まれて表示される）
CELLS[17] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.1, 110], [1.7, 110], [2.5, 225], [3.5, 225], [4.5, 0]], t), y = SY(u);
  const COLX = [15, 180, 345], IW = 151, IH = 92, ROWS = [12, 168, 276], WAIT = 0.45, FADE = 0.35;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'ギャラリー', 15, 140, `800 18px ${JP}`, C.ink); bars(k, 120, 128, 200, 1);
    ROWS.forEach((top, r) => COLX.forEach((x, c) => {
      const t0 = r === 0 ? -1e9 : enteredAt(SY, top, VH, u), age = t0 === null ? -1 : u - t0 - c * 0.12;
      k.fillStyle = C.skel; rr(k, x, top, IW, IH, 6); k.fill();
      if (age < WAIT + FADE) {   // まだの画像: 灰色の枠と画像の印
        k.save(); k.translate(x + IW / 2, top + IH / 2); k.strokeStyle = '#B9C1CF'; k.lineWidth = 2; k.lineJoin = 'round';
        if (age >= 0 && age < WAIT) spinner(k, 0, 0, 11, u, C.blue);
        else { rr(k, -15, -11, 30, 22, 3); k.stroke(); k.beginPath(); k.moveTo(-11, 7); k.lineTo(-3, -1); k.lineTo(3, 5); k.lineTo(7, 1); k.lineTo(11, 6); k.stroke(); }
        k.restore();
      }
      if (age >= WAIT) { k.save(); k.globalAlpha = eOut(P(age, WAIT, WAIT + FADE)); pic(k, x, top, IW, IH, (r + c) % 3, 6); k.restore(); }
    }));
    k.restore();
    sbar(k, y / 225, 0.45);
  });
};
