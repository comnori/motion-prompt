// 스크롤 타임라인 / Timeline on scroll
// 팩: lp2 (랜딩 페이지와 웹 효과 2) · element 01 · 분류: 스크롤 효과
// 동작 요약: 스크롤하면 타임라인의 주황색 선이 아래로 자랍니다. 선이 닿은 점이 채워지고 연도와 사건이 오른쪽에서 나타나며, 위로 스크롤하면 되돌아갑니다.
// 동작 사양:
//   - 511 × 168 뷰포트. 스크롤: 0.3–2.3 s에 0→190 px, 3.0–4.3 s에 190→0(cubic ease-in-out). 한 번은 4.7 s(팩의 9.4 s 주기마다 두 번)입니다.
//   - 타임라인은 x = 168입니다. 회색 트랙(두께 3) 위로 주황 선이 자랍니다. 끝은 scroll + 126 px이므로 항상 뷰포트 위에서 126 px(아래에서 42 px)에 놓입니다.
//   - 항목 다섯 개(2018–2026)의 점(흰색, 반지름 7.5, 고리 두께 2.5)을 페이지 y = 72, 126, 180, 234, 288에 놓습니다.
//   - 끝이 점을 지나고 선이 22 px 더 자라는 동안(cubic ease-out) 고리는 주황색이 되고 안쪽 점 반지름은 0에서 4로 커집니다. 연도(18 px, 주황)와 사건(15 px)은 오른쪽 16 px에서 이동하며 나타납니다.
//   - 되돌아 스크롤하면 같은 식으로 선을 줄이고 항목을 숨깁니다. 오른쪽 스크롤바 손잡이도 따라갑니다.
// 활용 상황: 회사 연혁·주요 성과·절차를 시간순으로 읽게 할 때 사용합니다.
// 태그: timeline, scroll, history, milestones, scroll-driven, progress-line
// 크기: 539x208 px · 반복 주기: 9.4 s
// 필요한 코드: kit lp2 -> https://comnori.github.io/motion-prompt/api/v1/kit/lp2.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#145
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 01
// 01 タイムラインアニメーション（スクロールに合わせて年表の線が上から伸び、線が届いた年が順に現れる）
CELLS[0] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const y = steps([[0.3, 0], [2.3, 190], [3.0, 190], [4.3, 0]], u);
  const ROWS = [[72, '2018', '創業'], [126, '2020', '1号店オープン'], [180, '2022', '会員1万人'], [234, '2024', 'アプリ公開'], [288, '2026', '海外へ']];
  const LX = 168, L = y + 126;                         // 線の先（画面の下から 42px の所までページと一緒に伸びる）
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'HISTORY', 34, 34, `800 11px ${EN}`, C.acc, 'left', 'alphabetic', '2px');
    txt(k, 'あゆみ', 34, 60, `800 20px ${JP}`, C.ink);
    pic(k, 34, 76, 100, 62, 0, 6); bars(k, 34, 148, 90, 2);
    k.fillStyle = C.faint; rr(k, LX - 1.5, 52, 3, 248, 1.5); k.fill();
    k.fillStyle = C.acc; rr(k, LX - 1.5, 52, 3, clamp(L - 52, 0, 248), 1.5); k.fill();
    for (const [iy, yr, lab] of ROWS) {
      const q = eOut(clamp((L - iy) / 22));
      k.fillStyle = '#fff'; k.beginPath(); k.arc(LX, iy, 7.5, 0, 7); k.fill();
      k.lineWidth = 2.5; k.strokeStyle = q > 0 ? C.acc : C.faint; k.stroke();
      if (q > 0) { k.fillStyle = C.acc; k.beginPath(); k.arc(LX, iy, 4 * q, 0, 7); k.fill(); }
      k.save(); k.globalAlpha = q; k.translate(16 * (1 - q), 0);
      txt(k, yr, LX + 22, iy + 6, `800 18px ${EN}`, C.acc);
      txt(k, lab, LX + 82, iy + 5, `700 15px ${JP}`, C.ink);
      bars(k, LX + 82, iy + 15, 190, 1, C.faint, 5);
      k.restore();
    }
    k.restore();
    sbar(k, y / 190, 0.45);
  });
};
