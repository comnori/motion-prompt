const ISO = (px, py) => [(px - py) * 0.866, (px + py) * 0.5];
CELLS[22] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const X = 200, Y = 156, SP = 30, SC = 0.85;
  const ex = i => backOut(P(u, 0.45 + i * 0.08, 1.2 + i * 0.08), 1.3) * (1 - eIO(P(u, 3.25 + (3 - i) * 0.07, 3.95 + (3 - i) * 0.07)));
