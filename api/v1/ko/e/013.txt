// 리퀴드 모션 / Liquid motion
// 팩: motion (모션 그래픽 움직임) · element 13 · 분류: 움직임 다듬기
// 동작 요약: 덩어리가 길게 늘어나 물방울로 나뉘었다가 다시 하나로 합쳐집니다.
// 동작 사양:
//   - 선두 원(반지름 44): 0.35–1.45 s에 cubic-bezier(.7, 0, .2, 1)로 오른쪽 300 px 이동하며 사인 곡선을 따라 34 px 올라갑니다. 2.7–3.8 s에 같은 방식으로 돌아옵니다.
//   - 일곱 원이 같은 경로를 각각 0.065 s 늦게 따라갑니다. 반지름은 44→15로 줄고 위아래 ±6 px 흔들립니다.
//   - 멈춘 동안에도 작은 원 세 개(반지름 15, 13, 11)가 주기마다 두 번, 거리 34–64 px 사이를 오갑니다.
//   - 합쳐지는 표현: 원을 흰색으로 그림 → 9 px 블러 → 알파 임곗값 적용(112–144 사이만 부드럽게) → 남은 영역을 색으로 채웁니다.
//   - 민트 → 파랑 → 보라(#7C5CFF)의 대각선 그라데이션으로 채우고 선두 원에 흰 타원 하이라이트를 넣습니다.
// 활용 상황: 로고·배경·연결 동작에 유기적이고 부드러운 느낌을 줄 때 사용합니다.
// 태그: liquid, blob, goo, gooey, metaball, fluid, organic, drip
// 크기: 539x208 px · 반복 주기: 4.7 s
// 필요한 코드: kit motion -> https://comnori.github.io/motion-prompt/api/v1/kit/motion.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#013
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 13
// 13 リキッドモーション（しずくが伸びて、ちぎれて、ぬるっと1つにまとまる）
const LQ = bez(.7, 0, .2, 1);
const lqPos = uu => { const u = mod(uu, CYC); let p; if (u < 0.35) p = 0; else if (u < 1.45) p = LQ(P(u, 0.35, 1.45)); else if (u < 2.7) p = 1; else if (u < 3.8) p = 1 - LQ(P(u, 2.7, 3.8)); else p = 0; const mvp = (u >= 0.35 && u < 1.45) ? P(u, 0.35, 1.45) : (u >= 2.7 && u < 3.8) ? P(u, 2.7, 3.8) : 0; return [lerp(120, 420, p), 104 - 34 * Math.sin(Math.PI * mvp)]; };
CELLS[12] = (k, w, h, u) => {
  stage(k);
  const blobs = [];
  for (let i = 0; i < 8; i++) { const [x, y] = lqPos(u - i * 0.065); blobs.push([x, y + (i ? 6 * Math.sin(u * 2.67 + i) : 0), lerp(44, 15, i / 7)]); }
  // 止まっている間、まわりで小さなしずくが出入りする
  for (let j = 0; j < 3; j++) { const [x, y] = lqPos(u - 0.1), a = j * 2.2 + 0.6, d = 34 + 30 * (0.5 - 0.5 * Math.cos(u / CYC * Math.PI * 4 + j * 2.1)); blobs.push([x + Math.cos(a) * d, y + Math.sin(a) * d * 0.8, 15 - j * 2]); }
  const L = goo(g => { for (const b of blobs) { g.beginPath(); g.arc(b[0], b[1], b[2], 0, 7); g.fill(); } }, 9, g => { const gr = g.createLinearGradient(60, 30, 480, 180); gr.addColorStop(0, C.mint); gr.addColorStop(0.55, C.blue); gr.addColorStop(1, '#7C5CFF'); g.fillStyle = gr; g.fillRect(0, 0, w, h); });
  k.drawImage(L, 0, 0);
  const [hx, hy] = lqPos(u); k.fillStyle = 'rgba(255,255,255,.75)'; k.beginPath(); k.ellipse(hx - 14, hy - 17, 10, 6, -0.6, 0, 7); k.fill();
};
