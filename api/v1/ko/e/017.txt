// 동작 사양:
//   - 낮 → 밤 → 낮 장면을 각각 2.35 s씩 표시합니다. 각 절반의 0.45–1.6 s(1.15 s) 동안 cubic ease-in-out으로 전환합니다.
//   - 다음 장면을 현재 장면 위에 도형으로 클리핑해 그립니다. 도형은 프레임 전체를 덮을 때까지 커집니다.
//   - 낮→밤: 태양 중심(112, 104)에서 원 반지름 0→470.
//   - 밤→낮: 달 중심(427, 104)에서 오각별의 바깥 반지름 0→1150. 커지면서 1.2 rad(약 69°) 회전합니다.
//   - 커지는 동안 도형 가장자리에 두께 7의 흰 윤곽선을 표시합니다.
// 활용 상황: 태양·로고·버튼처럼 화면 안의 물체를 출발점으로 장면을 전환할 때 사용합니다.
// 태그: mask, transition, iris, wipe, shape-wipe, reveal, scene-change
// 크기: 539x208 px · 반복 주기: 4.7 s
// 필요한 코드: kit motion -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion.js

// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 17
// 17 マスクトランジション（丸や星の形の「窓」が広がり、その中に次の画面が見える）
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
