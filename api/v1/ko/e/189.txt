// 동작 요약: 이야기하는 사람에서 관객의 반응으로 잠시 전환한 뒤 원래 장면으로 돌아옵니다.
// 동작 사양:
//   - 1.7 s에 다른 장면으로, 3.0 s에 원래 장면으로 전환 프레임 0개의 컷을 사용합니다. 다른 장면은 1.3 s 동안 보입니다.
//   - 장면 A: 카메라 높이 1.4 m로 고정. 거리 5.5 m의 사람이 말하며 팔다리를 0.45·sin(u×6) rad로 움직입니다. 회색 연단은 오른쪽 0.9 m에 있습니다.
//   - 관객 장면에는 거리 3.1–4.6 m에 파란색과 연한 파란색 사람 다섯 명을 배치합니다. 팔을 들고 |sin(t×9+i)|×5 cm만큼 뜁니다. 배경을 공유하지 않는 다른 장소입니다.
//   - 하단은 회색 A → 파란 다른 장면 → 회색 A이며 재생 헤드가 이동합니다. 17 Insert는 같은 장면의 세부를 보여 줍니다.
//   - 상단은 ‘SHOT A → CUTAWAY → SHOT A’입니다. 4.7 s에 시작 화면으로 컷합니다.
// 활용 상황: 반응을 보여 주거나 원래 장면의 생략·말실수를 가릴 때 사용합니다.
// 태그: cutaway, reaction shot, b-roll, cut away, audience, edit
// 크기: 539x208 px · 반복 주기: 9.4 s
// 필요한 코드: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#189
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 21
// 21 カットアウェイ: 話す人物から、別の対象（聞いている客席）へ切り替えて戻る。17 インサートは同じ場面の細部
const CUT = [1.7, 3.0];
CELLS[20] = shot((k, w, h, u) => {
  const away = u >= CUT[0] && u < CUT[1];
  if (!away) world(k, w, h, cam({ y: 1.4 }), { person: { x: 0, z: 5.5, walk: 0.45 * Math.sin(u * 6) }, extra: [[0.9, 0, 5.6, (kk) => { kk.fillStyle = C.g3; kk.fillRect(-0.3, 0, 0.6, 1.1); kk.fillRect(-0.4, 1.05, 0.8, 0.1); }]] });
  else {
    const t = u - CUT[0], ex = [];
    [[-1.6, 3.2], [-0.2, 3.6], [1.3, 3.1], [0.6, 4.4], [-1.1, 4.6]].forEach(([x, z], i) => ex.push([x, 0, z, (kk) => { kk.save(); kk.translate(0, Math.abs(Math.sin(t * 9 + i)) * 0.05); person(kk, 1, { armUp: true, col: i % 2 ? C.blue : C.blueM }); kk.restore(); }]));
    world(k, w, h, cam({ y: 1.6, pitch: -0.08 }), { buildings: false, trees: false, mountains: false, sun: null, extra: ex });
  }
  strip(k, w, h, [[0, CUT[0], 'A', C.g2], [CUT[0], CUT[1], '別の対象', C.blue], [CUT[1], CYC, 'A', C.g2]], u);
  return away ? 'CUTAWAY' : 'SHOT A';
});
