// 동작 요약: 세 제목 중 첫 번째를 누르면 내용이 펼쳐지고 아래 행이 밀려납니다. 다시 누르면 접힙니다.
// 동작 사양:
//   - 542×191 카드에서 4.5 s 주기. 제목 세 개(너비 494, 높이 34, 모서리 반지름 8, 간격 6)의 오른쪽에 아래 화살표를 둡니다.
//   - 커서는 0.95 s에 첫 제목에 도착하고 1.0 s와 3.3 s에 클릭합니다.
//   - 열림 정도 ex: 1.0–1.35 s에 easeInOutCubic으로 0→1, 3.3–3.62 s에 1→0.
//   - 첫 행의 높이는 34에서 92로 증가(+58·ex)하여 아래 행을 밀어냅니다. 내용 두 줄은 행 영역으로 클리핑하고 불투명도 ex로 나타냅니다.
//   - 열리는 동안 화살표를 180°·ex 회전하고 테두리를 옅은 파랑, 제목을 파랑으로 바꿉니다.
// 활용 상황: FAQ나 긴 설명에서 제목을 훑어보고 필요한 내용만 펼치게 할 때 사용합니다.
// 태그: accordion, collapsible, expand, collapse, disclosure, faq, details
// 크기: 542x191 px · 반복 주기: 4.5 s
// 필요한 코드: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#052
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 04
// 04 アコーディオン（Accordion）
CELLS[3] = ui((w, h, t) => {
  const ex = eio(seg(t, 1.0, 1.35)) * (1 - eio(seg(t, 3.3, 3.62)));
  let y = 12; const H0 = 34, EX = 58;
  for (let i = 0; i < 3; i++) {
    const extra = i === 0 ? EX * ex : 0;
    strokeRR(24, y, 494, H0 + extra, 8, i === 0 && ex > .05 ? C.sb : C.line2, 1.5);
    bar(42, y + 12, [180, 220, 150][i], i === 0 && ex > .05 ? C.blue : C.line2, 10);
    chevron(494, y + 17, 6, i === 0 ? Math.PI * ex : 0, i === 0 && ex > .05 ? C.blue : '#9a9896');
    if (i === 0 && ex > 0) { X.save(); rr(24, y, 494, H0 + extra, 8); X.clip(); X.globalAlpha = ex; bar(42, y + H0 + 6, 420); bar(42, y + H0 + 24, 320); X.restore(); }
    y += H0 + extra + 6;
  }
  pointer([[.3, w + 30, h + 30], [.95, 300, 31], [1.6, 300, 31], [3.0, 300, 31], [3.55, 300, 31], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
