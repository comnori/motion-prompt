// 토스트 알림 / Toast notification
// 팩: ui (UI 컴포넌트 이름) · element 08 · 분류: 펼치기와 오버레이
// 동작 요약: “저장”을 누르면 어두운 알림이 오른쪽 아래로 들어오고, 남은 시간 막대가 끝나면 빠져나갑니다.
// 동작 사양:
//   - 542×191 카드에서 4.5 s 주기. 1.0 s에 왼쪽 아래 파란 “저장” 버튼(110×36)을 클릭한 뒤 커서가 왼쪽 아래로 나갑니다.
//   - 토스트: #26282c, 214×52, 모서리 반지름 10, 오른쪽에서 18 px·아래에서 16 px. 1.05–1.4 s에 easeOutCubic으로 오른쪽 240 px에서 들어옵니다.
//   - 내부: 파란 원 안의 흰 체크와 15 px 흰 글자 “저장됨”.
//   - 하단의 남은 시간 선(높이 3 px)이 1.4–3.2 s에 선형으로 194 px에서 0으로 줄어듭니다.
//   - 퇴장: 3.2–3.55 s에 easeInOutCubic으로 오른쪽 240 px 밖으로 이동합니다.
// 활용 상황: 사용자를 방해하지 않고 저장·전송 등의 결과를 알려 줄 때 사용합니다.
// 태그: toast, snackbar, notification, alert, message, auto-dismiss
// 크기: 542x191 px · 반복 주기: 4.5 s
// 필요한 코드: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#056
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 08
// 08 トースト（Toast notification）
CELLS[7] = ui((w, h, t) => {
  bar(24, 16, 90, C.line2, 10); strokeRR(24, 34, 300, 34, 7, C.line2, 1.5); bar(24, 82, 90, C.line2, 10); strokeRR(24, 100, 300, 34, 7, C.line2, 1.5);
  fillRR(24, 146, 110, 36, 8, C.blue); text('保存', 79, 164, 16, '#fff', 600, 'center');
  const tin = eo(seg(t, 1.05, 1.4)), tout = eio(seg(t, 3.2, 3.55));
  if (tin > 0 && tout < 1) {
    const tw0 = 214, th = 52, tx = w - 18 - tw0 + (1 - tin) * 240 + tout * 240, ty = h - 16 - th;
    shadowed(() => fillRR(tx, ty, tw0, th, 10, C.dark), 18, 6, .25);
    circle(tx + 26, ty + 24, 11, C.blue); check(tx + 26, ty + 24, 5, '#fff', 2.4);
    text('保存しました', tx + 46, ty + 24, 15, '#fff', 600);
    const life = 1 - seg(t, 1.4, 3.2); fillRR(tx + 10, ty + th - 7, (tw0 - 20) * life, 3, 1.5, C.sb);
  }
  pointer([[.3, -30, h + 30], [.95, 84, 168], [1.5, 84, 168], [2.2, -40, h + 40]], [1.0], t);
});
