// 팩: photo (사진 표현·조명·스타일) · element 10 · 분류: 렌즈와 카메라 효과
// 동작 요약: 거리의 위아래는 흐려지고 도로·자동차·작은 사람들이 있는 가운데 띠만 선명하게 남아 모형처럼 보입니다.
// 동작 사양:
//   - 큰 인물이 없는 거리 모퉁이입니다(건물·나무·가로등 뒤 도로 y 132–162). 키 23–26 px의 작은 사람 네 명이 보도에 서 있습니다(발 y 172). 자동차는 5 s마다 x −60에서 600으로 이동합니다.
//   - 초점 띠: y 66–176은 선명하게 유지합니다.
//   - 블러: y 0–42와 y 190–208은 0 → 7 px, y 42–66과 y 176–190은 0 → 3 px.
//   - 띠 경계에 점선을 표시합니다(선 6 px, 간격 5 px, #8E8E93, 75%).
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 도시나 행사를 장난감 모형 세계처럼 보이게 할 때 사용합니다.
// 태그: tilt-shift, miniature, miniature effect, fake miniature, selective focus, toy town, focus band
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#106
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 10
// ティルトシフト: 上下の帯がぼけて中央の帯だけピントが合い、街と小さな人と車が模型のように見える
CELLS[9] = cell((k, u) => {
  const a = env(u), carX = lerp(-60, 600, u / CYC);
  const minis = D => { [[196, 26], [322, 26], [340, 23], [452, 25]].forEach(([x, h]) => figure(D, x, 172, h, BASE.person)); };
  const [S, ks] = layer('tilt'); stageStreet(ks, BASE, { car: carX, person: false, after: minis });
  const [B1, k1] = layer('tilt-1'); k1.filter = `blur(${(3 * a).toFixed(2)}px)`; k1.drawImage(S, 0, 0); k1.filter = 'none';
  const [B2, k2] = layer('tilt-2'); k2.filter = `blur(${(7 * a).toFixed(2)}px)`; k2.drawImage(S, 0, 0); k2.filter = 'none';
  k.drawImage(S, 0, 0);
  const band = (img, y0, y1) => { k.save(); k.beginPath(); k.rect(0, y0, CW, y1 - y0); k.clip(); k.drawImage(img, 0, 0); k.restore(); };
  band(B2, 0, 42); band(B1, 42, 66); band(B1, 176, 190); band(B2, 190, CH);
  // ピントの帯の目印（写真アプリの線）
  k.save(); k.globalAlpha = 0.75 * a; k.strokeStyle = '#8E8E93'; k.lineWidth = 1.4; k.setLineDash([6, 5]);
  k.beginPath(); k.moveTo(0, 66); k.lineTo(CW, 66); k.moveTo(0, 176); k.lineTo(CW, 176); k.stroke(); k.setLineDash([]);
  k.restore();
});
