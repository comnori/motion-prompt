// 렌즈 플레어 / Lens flare
// 팩: photo (사진 표현·조명·스타일) · element 13 · 분류: 렌즈와 카메라 효과
// 동작 요약: 해가 건물 뒤에서 나오면 주변에 광선이 생기고 해에서 화면 중심을 지나 반대편으로 빛점·육각형·고리가 늘어섭니다.
// 동작 사양:
//   - 장면 2: 거리 모퉁이. y 132–162의 도로 뒤에 건물·나무·가로등(x 114, 292, 494)이 있습니다. 왼쪽 앞에 사람이 서 있습니다(발 x 176, y 200, 키 124 px).
//   - 해: 반지름 13 px, 건물 뒤(x 428, y 74)에서 열린 하늘(x 474, y 30)로 이동합니다.
//   - 광선: 8개, 해 중심에서 17 px 떨어진 곳부터 26 px 또는 44 px까지 교대합니다(길이 9 px·27 px, 호박색, 두께 1.6 px, 75%).
//   - 플레어: 해에서 중심을 지나는 선의 t = 0.45, 0.8, 1.15, 1.45, 1.75, 2.0에 놓습니다(t = 1은 중심). 7 px 원, 13 px 육각형, 5 px 원, 22 px 육각형, 9 px 원, 30 px 고리(선 3 px).
//   - 색은 호박색 #FFC75A와 #FFE3A3이며 불투명도 40–70% × 효과입니다.
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 눈부심·여름 햇살·영화 같은 빛의 느낌을 더할 때 사용합니다.
// 태그: lens flare, flare, ghosting, sun flare, starburst, cinematic
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://comnori.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#109
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 13
// レンズフレア: 建物の陰から太陽が出ると、画面の中心をはさんで反対側へ光の玉が並ぶ
CELLS[12] = cell((k, u) => {
  const a = env(u), sx = lerp(428, 474, a), sy = lerp(74, 30, a);
  stageStreet(k, BASE, { sun: [sx, sy, 13, 1] });
  const cx = CW / 2, cy = CH / 2, dx = cx - sx, dy = cy - sy;
  k.save();
  // 太陽の光の筋
  k.globalAlpha = 0.75 * a; k.strokeStyle = BASE.lit; k.lineWidth = 1.6; k.lineCap = 'round';
  for (let i = 0; i < 8; i++) { const t = i / 8 * TAU + 0.2, L = i % 2 ? 26 : 44; k.beginPath(); k.moveTo(sx + Math.cos(t) * 17, sy + Math.sin(t) * 17); k.lineTo(sx + Math.cos(t) * L, sy + Math.sin(t) * L); k.stroke(); }
  // 光の玉: 太陽から中心を通る線の上（t = 0.45〜2.0）
  const G = [[0.45, 7, 'c', 0.6], [0.8, 13, 'h', 0.55], [1.15, 5, 'c', 0.7], [1.45, 22, 'h', 0.4], [1.75, 9, 'c', 0.55], [2.0, 30, 'r', 0.55]];
  for (const [t, r, kind, al] of G) {
    const x = sx + dx * t, y = sy + dy * t;
    k.globalAlpha = al * a;
    if (kind === 'r') { k.strokeStyle = BASE.lit; k.lineWidth = 3; k.beginPath(); k.arc(x, y, r, 0, TAU); k.stroke(); continue; }
    k.fillStyle = kind === 'h' ? BASE.glow : BASE.lit; k.beginPath();
    if (kind === 'h') for (let i = 0; i < 6; i++) { const t2 = i / 6 * TAU; k.lineTo(x + Math.cos(t2) * r, y + Math.sin(t2) * r); } else k.arc(x, y, r, 0, TAU);
    k.closePath(); k.fill();
  }
  k.restore();
});
