// 가로 스크롤 / Horizontal scroll
// 팩: lp (랜딩 페이지와 웹 효과) · element 05 · 분류: 스크롤 효과
// 동작 요약: 아래로 스크롤하면 섹션은 고정되고 번호가 붙은 작업 카드 네 개가 왼쪽으로 이동합니다.
// 동작 사양:
//   - 스크롤: 0.3–3.0 s에 0→420 px, 3.5–4.5 s에 420→0(cubic ease-in-out).
//   - “WORKS” 제목은 위로 사라지고 섹션은 스크롤 40 px에서 상단에 닿아 고정됩니다.
//   - 카드: 200 × 112 네 개, 간격 20, 양옆 여백 34. 가로 이동량 = 34·2 + 200·4 + 20·3 − 511 = 417 px.
//   - 카드 x = 시작 x − p × 417, p = (scroll − 40) / 380. 스크롤에 선형 대응하며 이징은 없습니다.
//   - 카드가 옆으로 이동하는 동안 스크롤바 손잡이는 내려갑니다.
// 활용 상황: 긴 세로 페이지 안에서 작업물·사례·제품을 가로로 나열할 때 사용합니다.
// 태그: horizontal-scroll, sideways, scrolltrigger, pin, gallery, portfolio, scroll
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#077
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 05
// 05 横スクロール（縦にスクロールすると、カードの列が横へ流れる）
CELLS[4] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.3, 0], [3.0, 420], [3.5, 420], [4.5, 0]], u);
  const PIN0 = 40, PD = 380, N = 4, PWD = 200, GAP = 20;
  const top = y < PIN0 ? PIN0 - y : 0, p = clamp((y - PIN0) / PD);
  const travel = 34 * 2 + N * PWD + (N - 1) * GAP - VW;
  view(k, () => {
    txt(k, 'WORKS', 34, 30 - y, `800 20px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
    k.save(); k.translate(0, top);
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH + 40);
    for (let i = 0; i < N; i++) {
      const x = 34 + i * (PWD + GAP) - p * travel;
      pic(k, x, 18, PWD, 112, i % 3, 8);
      txt(k, '0' + (i + 1), x + 14, 118, `800 22px ${EN}`, '#fff');
      bars(k, x, 140, 120, 1, '#CBD2DE');
    }
    k.restore();
    sbar(k, y / 420, 0.36);
  });
};
