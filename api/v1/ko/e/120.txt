// 이중 노출 / Double exposure
// 팩: photo (사진 표현·조명·스타일) · element 24 · 분류: 색조와 마무리 표현
// 동작 요약: 배경은 흰색이 되고 언덕·집·나무·해의 풍경이 작아져 흉상 모양 안에만 나타납니다.
// 동작 사양:
//   - 장면 3: 왼쪽에 흉상(머리 중심 x 140, y 82, 반지름 28 px), 오른쪽에 언덕·집·나무·해(x 432, y 54, 반지름 15 px)가 있습니다.
//   - 배경: 흉상 바깥은 흰색 #FFFFFF.
//   - 흉상 안: 같은 풍경을 0.7×로 축소합니다. 해는 머리 오른쪽 위에, 언덕 능선은 목을 가로지르도록 배치합니다.
//   - 내부 색: 하늘 #D3DBE6, 언덕 #9AA8BD·#76869F, 지면 #55647E, 나무 #3F4C66.
//   - 전환: 이중 노출 이미지를 효과 양에 따라 원본 위에 겹칩니다.
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 인물을 장소·기억·내면세계와 연결하는 포스터나 표지에 사용합니다.
// 태그: double exposure, multiple exposure, blend, silhouette, surreal, composite
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://comnori.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#120
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 24
// 二重露光: 背景が白く飛び、胸像の形の中にだけ丘と家と木と太陽が写り込む
CELLS[23] = cell((k, u) => {
  const a = env(u);
  const DX = Object.assign({}, BASE, { sky: '#D3DBE6', sky2: '#D3DBE6', far: '#9AA8BD', mid: '#76869F', ground: '#55647E', bld: '#E8ECF2', edge: '#55647E', win: '#9AA8BD', winOff: '#9AA8BD', tree: '#3F4C66', trunk: '#3F4C66' });
  const draw = kk => {
    kk.fillStyle = '#FFFFFF'; kk.fillRect(0, 0, CW, CH);
    kk.save(); bustPath(kk, QX, QY, QR); kk.clip();
    // 風景を0.7倍にして、太陽が頭の右上、丘の稜線が首のあたりに来るように置く
    kk.translate(QX, QY); kk.scale(0.7, 0.7); kk.translate(-415, -77);
    kk.fillStyle = DX.sky; kk.fillRect(0, -200, CW * 2, CH * 3);
    stageLand(kk, DX, { person: false });
    kk.fillStyle = DX.ground; kk.fillRect(-CW, 200, CW * 3, CH * 3);
    kk.restore();
  };
  if (a >= 0.999) { draw(k); return; }   // 効果が入りきったら直接描く（拡大表示でも輪郭がくっきり）
  stageLand(k, BASE);
  if (a < 0.002) return;
  const [L, kl] = layer('dx'); draw(kl);   // 入り・戻りの途中は、別の面に描いて重ねる
  k.save(); k.globalAlpha = a; k.drawImage(L, 0, 0); k.restore();
});
