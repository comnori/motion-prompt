// 동작 요약: 네이비 화면의 로고 아래 숫자가 0%에서 100%로 오른 뒤 화면이 위로 걷히고 아래 페이지가 나타납니다.
// 동작 사양:
//   - 네이비 화면이 뷰포트를 채우며 로고·숫자(28 px)·막대(180 × 4)를 표시합니다.
//   - 0.25–2.1 s에 0→100%(cubic ease-out, 정수 반올림). 막대도 같은 비율로 주황색이 채워집니다.
//   - 2.3–2.85 s에 화면 전체가 168 px 위로 이동해 나갑니다(cubic ease-in-out).
//   - 페이지 제목과 사진은 2.35–2.95 s에 불투명도 0→1, 아래 14 px에서 위로 올라오며 나타납니다(cubic ease-out).
//   - 4.25–4.75 s에 다음 로딩을 나타내는 0% 화면이 위에서 내려옵니다.
// 활용 상황: 이미지나 영상이 많은 사이트에서 로딩 시간을 브랜드 표현에 활용할 때 사용합니다.
// 태그: preloader, loading-screen, loader, percentage, intro, splash, curtain
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#089
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 17
// 17 ローディング画面（数字が100%まで進むと、幕が上がってページが現れる）
CELLS[16] = (k, w, h, u) => {
  frame(k);
  const n = Math.round(100 * eOut(P(u, 0.25, 2.1)));
  const cy = u < 4.25 ? -VH * eIO(P(u, 2.3, 2.85)) : -VH * (1 - eIO(P(u, 4.25, 4.75)));
  const show = eOut(P(u, 2.35, 2.95));
  view(k, () => {
    header(k, 0);
    k.save(); k.globalAlpha = show; k.translate(0, 14 * (1 - show));
    txt(k, 'ようこそ。', 34, 82, `800 26px ${JP}`, C.ink); bars(k, 34, 98, 200, 3); btn(k, 34, 136, 96, 24, '始める');
    pic(k, 270, 46, 214, 112, 0, 8); k.restore();
    if (cy > -VH) {
      k.save(); k.translate(0, cy); k.fillStyle = C.ink; k.fillRect(0, 0, VW, VH);
      logo(k, VW / 2 - 40, 56, '#fff');
      txt(k, (u >= 4.25 ? 0 : n) + '%', VW / 2, 98, `800 28px ${EN}`, '#fff', 'center', 'middle');
      k.fillStyle = 'rgba(255,255,255,.15)'; rr(k, VW / 2 - 90, 122, 180, 4, 2); k.fill();
      k.fillStyle = C.acc; rr(k, VW / 2 - 90, 122, 180 * (u >= 4.25 ? 0 : n / 100), 4, 2); k.fill();
      k.restore();
    }
  });
};
