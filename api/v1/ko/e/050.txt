// 동작 요약: “메뉴”를 클릭하면 왼쪽에서 긴 패널이 들어오고, 바깥을 클릭하면 다시 들어갑니다.
// 동작 사양:
//   - 542×191 카드에서 4.5 s 주기. 커서가 1.0 s에 왼쪽 위 “메뉴” 버튼(110×34)을 클릭합니다.
//   - 열림 정도 a: 1.0–1.38 s에 easeOutCubic으로 0→1. 카드 전체의 rgba(18,20,28) 덮개 불투명도는 0.36·a입니다.
//   - 패널: 흰색, 너비 210, 높이는 카드 전체. x = −210 + 210·a로 왼쪽에서 들어옵니다. 그림자 블러 22, 알파 0.25.
//   - 내부: 위에 아바타(반지름 14)와 이름, 그 아래 32 px 간격의 메뉴 네 줄. 첫 줄은 옅은 파란 띠로 선택 표시합니다.
//   - 닫기: 3.3 s에 커서가 바깥 (420, 120)을 클릭합니다. a에 (1 − 3.3–3.65 s의 easeInOutCubic)을 곱해 패널을 되돌립니다.
// 활용 상황: 특히 모바일에서 탐색 메뉴나 필터를 필요할 때만 꺼내 쓸 수 있게 할 때 사용합니다.
// 태그: drawer, off-canvas, offcanvas, side-menu, navigation-drawer, sheet, sidebar, slide-in
// 크기: 542x191 px · 반복 주기: 4.5 s
// 필요한 코드: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#050
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 02
// 02 ドロワー（Drawer (Off-canvas)）
CELLS[1] = ui((w, h, t) => {
  strokeRR(24, 14, 110, 34, 8, C.line2, 1.5); text('メニュー', 79, 31, 15, '#555', 600, 'center');
  bar(160, 26, 140, C.line2, 11); pageLines(24, 70, 494); fillRR(24, 138, 494, 42, 8, C.block);
  const a = eo(seg(t, 1.0, 1.38)) * (1 - eio(seg(t, 3.3, 3.65)));
  if (a > 0) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const pw = 210, px = -pw + pw * a;
    shadowed(() => { X.fillStyle = '#fff'; X.fillRect(px, 0, pw, h); }, 22, 0, .25);
    circle(px + 34, 32, 14, C.mb); bar(px + 58, 27, 100, C.line2, 10);
    for (let i = 0; i < 4; i++) { const y = 62 + i * 32; if (i === 0) fillRR(px + 12, y - 2, pw - 24, 28, 6, C.lb); fillRR(px + 22, y + 5, 14, 14, 3, i === 0 ? C.blue : C.line2); bar(px + 46, y + 7, 90 + (i % 2) * 30, i === 0 ? C.sb : C.line); }
  }
  pointer([[.3, w + 30, h + 30], [.95, 70, 34], [1.6, 70, 34], [3.0, 420, 120], [3.6, 420, 120], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
