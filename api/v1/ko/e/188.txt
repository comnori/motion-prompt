// 아이리스 아웃 / Iris out
// 팩: camera2 (카메라 움직임과 장면 전환 2) · element 20 · 분류: 전환과 편집
// 동작 요약: 원형 창이 닫히며 화면이 검게 되고, 다음 피사체를 중심으로 다시 열립니다.
// 동작 사양:
//   - 아이리스 아웃은 1.3–2.7 s(1.4 s), 반지름 340 → 0 px입니다. quadratic ease-in-out, 약 cubic-bezier(0.45,0,0.55,1)을 사용하며 원 바깥은 #141418로 채웁니다.
//   - 원의 중심은 화면에 투영된 피사체 위치입니다. A의 사람 머리는 왼쪽 0.6 m, 높이 1.63 m, 거리 7 m입니다. B의 자동차는 오른쪽 1.2 m, 높이 0.9 m, 거리 9 m입니다.
//   - 2.7–3.1 s에는 검은 화면을 유지하고 2.9 s에 컷으로 장면을 바꿉니다.
//   - 아이리스 인은 3.1–4.4 s(1.3 s), 다음 피사체를 중심으로 반지름 0 → 340 px입니다.
//   - 반지름이 1 px 이상이면 2 px 파란 테두리를 표시합니다. 두 장면의 카메라는 고정합니다.
//   - 4.7 s마다 번갈아 전환하며 왕복은 9.4 s입니다. 상단 표시는 ‘IRIS OUT / IN’입니다.
// 활용 상황: 무성영화나 만화처럼 장난스러운 장면 전환에 사용합니다.
// 태그: iris out, iris in, iris wipe, circle, old film, cartoon, transition
// 크기: 539x208 px · 반복 주기: 9.4 s
// 필요한 코드: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#188
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 20
// 20 アイリスアウト: 人物を中心に丸い窓が閉じて黒になり、次の場面は丸い窓が開いて現れる
CELLS[19] = shot(abCell((k, w, h, u, A, B) => {
  const out = eIOs(P(u, 1.3, 2.7)), inn = eIOs(P(u, 3.1, 4.4));
  const first = u < 2.9, cur = first ? A : B;
  cur(k, w, h);
  // 窓の中心は、いま映っている場面の主役（Aは人物の頭、Bは車）
  const c = cur === sceneA ? proj(cam({ y: 1.4 }), -0.6, 1.63, 7) : proj(cam({ y: 1.6, pitch: 0.1 }), 1.2, 0.9, 9);
  const rad = first ? lerp(340, 0, out) : lerp(0, 340, inn);
  const cx = w / 2 + c.x, cy = h / 2 + c.y;
  if (rad < 339) {
    k.fillStyle = '#141418'; k.beginPath(); k.rect(0, 0, w, h); k.arc(cx, cy, Math.max(0, rad), 0, Math.PI * 2, true); k.fill('evenodd');
    if (rad > 1) { k.strokeStyle = C.blue; k.lineWidth = 2; k.beginPath(); k.arc(cx, cy, rad, 0, 7); k.stroke(); }
  }
}, u => u < 2.9 ? 'IRIS OUT' : 'IRIS IN'));
