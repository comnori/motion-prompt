// 골든 아워 / Golden hour
// 팩: photo (사진 표현·조명·스타일) · element 07 · 분류: 조명과 시간대
// 동작 요약: 해가 수평선 바로 위로 내려가고 하늘과 땅이 주황빛으로 바뀝니다. 인물 왼쪽에 빛이 들고 가로등과 나무 그림자가 오른쪽으로 길어집니다.
// 동작 사양:
//   - 장면 1: y 150 수평선 앞의 상반신 인물(머리 중심 x 270, y 78, 반지름 29 px, 얼굴·머리카락 없음).
//   - 해: 반지름 18 px, (x 110, y −30)에서 (x 96, y 136)으로 이동하며 언덕에 절반 가려집니다.
//   - 색: 하늘 위 #FAF9F7 → #FFE8BC, 수평선 부근 #FFCF80, 지면 #F1F0EE → #F6DDB0. 인물의 빛은 #FFB547, 그림자는 #3E3650입니다.
//   - 머리: 왼쪽 약 66°에서 빛을 비추며 경계 오른쪽은 그림자입니다. 몸의 왼쪽은 밝고 오른쪽은 어둡습니다.
//   - 그림자: 가로등과 나무 밑에서 오른쪽으로 길이 0 → 150 px(#DDAA6E). 오른쪽 위 시간은 12:00 → 17:40.
//   - 시간: 0.4 s부터 1.4 s까지 효과가 나타나고 4.0 s까지 유지한 뒤 4.8 s까지 사라집니다(한 번 = 5 s, cubic ease-in-out, 약 cubic-bezier(0.65,0,0.35,1)).
// 활용 상황: 따뜻하고 향수를 자아내는 부드러운 이미지에 사용합니다.
// 태그: golden hour, magic hour, sunset light, warm light, long shadows, low sun
// 크기: 539x208 px · 반복 주기: 5 s
// 필요한 코드: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 예제: https://showheyohtaki.github.io/motion-prompt/?lang=ko#103
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 07
// ゴールデンアワー: 太陽が地平線近くまで下り、全体が橙に染まって影が長く伸びる
CELLS[6] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  const GOLD = {
    sky: '#FFE8BC', sky2: '#FFCF80', far: '#F2C58C', mid: '#E8B27A', ground: '#F6DDB0', ground2: '#ECC892', mark: '#F6DDB0', cast: '#DDAA6E',
    bld: '#F7DCAE', edge: '#D9AE76', win: '#ECC892', winOff: '#ECC892', tree: '#E9C08A', trunk: '#C9965E', post: '#B98A58', bulb: '#C9965E',
    person: '#5C5470', shade: '#3E3650', lit: '#FFB547', car: '#A08E8E', glass: '#F6DDB0', tire: '#3E3650', sun: '#F59A24', glow: '#FFD48A'
  };
  const p = pal(BASE, GOLD, a);
  stagePortrait(k, p, { sun: [lerp(110, 96, a), lerp(-30, 136, a), 18, 1], cast: lerp(0, 150, a) });
  const lit = mixc(p.person, GOLD.lit, a), sh = mixc(p.person, GOLD.shade, a);
  bust(k, PX, PY, R, p.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    headShade(kk, PX, PY, R, 1.15, 1, sh);
    torso(kk, PX, PY, R, PX - W, PX - 0.5 * W, lit);
    torso(kk, PX, PY, R, PX + 0.4 * W, PX + W, sh);
  });
  hud(k, clock(lerp(720, 1060, a)));
});
