// 팩: camera (카메라 움직임과 장면 전환) · element 11 · 분류: 촬영 방식과 시간 연출
// 동작 요약: 두 줄의 나무 사이를 낮고 빠르게 날며 좌우로 기울어지는 1인칭 시점입니다.
// 동작 사양:
//   - 카메라: 일정한 14 m/s(50km/h로 표시)로 전진하며 이징은 없습니다. 피치는 -0.05 rad로 고정합니다.
//   - 좌우 이동(T = 2π×u/4.7, 4.7 s마다 두 번): 가로 0.6·sin(2T-0.4) m, 높이 1.4 + 0.5·sin(2T) m.
//   - 기울기: 롤 0.22·sin(2T+0.6) rad(약 ±12.6°), 요 0.05·sin(2T) rad(약 ±2.9°).
//   - 피사체 없음. 높이 4.2 m인 나무가 양옆 3.6 m에 8 m 간격으로 있고, 두 줄은 4 m 어긋나 있습니다. 나무가 카메라 쪽으로 빠르게 다가옵니다.
//   - 속도선: 반투명 회색 방사형 선 10개, 길이 약 32–76 px(양옆일수록 김). 바깥으로 흐르며 반복됩니다.
//   - 정지나 초기화 크로스페이드는 없습니다(4.7 s마다 다시 시작할 때 나무 줄의 위상이 바뀜).
// 활용 상황: 속도감과 몰입감을 주거나 한 번의 연속 이동으로 장소를 통과할 때 사용합니다.
// 태그: fpv, drone, fly through, first person, banking, speed, aerial
// 크기: 539x208 px · 반복 주기: 9.4 s
// 필요한 코드: kit camera -> https://comnori.github.io/motion-prompt/api/v1/kit/camera.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#035
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 11
// 11 FPV
CELLS[10] = shot((k, w, h, u) => {
  const sp = 14, z = u * sp, T = u * 2 * Math.PI / CYC;
  const c = cam({ z, y: 1.4 + 0.5 * Math.sin(T * 2), roll: 0.22 * Math.sin(T * 2 + 0.6), yaw: 0.05 * Math.sin(T * 2), x: 0.6 * Math.sin(T * 2 - 0.4), pitch: -0.05 });
  const ex = [];
  const L = 8, base = Math.floor(z / L) * L;
  for (let i = 0; i < 12; i++) { const zz = base + i * L; ex.push([-3.6, 0, zz, (kk, s) => tree(kk, s, 4.2)]); ex.push([3.6, 0, zz + L / 2, (kk, s) => tree(kk, s, 4.2, C.blueXL)]); }
  world(k, w, h, c, {
    buildings: false, trees: false, extra: ex, gridX: 20,
    after: (kk) => {   // スピード線
      kk.strokeStyle = 'rgba(150,150,155,.45)'; kk.lineWidth = 2;
      for (let i = 0; i < 10; i++) { const a = i / 10 * Math.PI * 2 + 0.3, ph = mod(u * 2.2 + i * 0.37, 1); const r0 = 120 + ph * 160; kk.beginPath(); kk.moveTo(Math.cos(a) * r0 * 1.9, Math.sin(a) * r0 * 0.8); kk.lineTo(Math.cos(a) * (r0 + 40) * 1.9, Math.sin(a) * (r0 + 40) * 0.8); kk.stroke(); }
    }
  });
  return `${Math.round(sp * 3.6)}km/h`;
});
