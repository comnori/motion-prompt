// 동작 요약: 커서 아래 사진만 천천히 커지고 틀은 그대로입니다. 커서를 치우면 사진이 줄어듭니다.
// 동작 사양:
//   - 사진 틀 두 개는 222 × 118, 모서리 반지름 8입니다. 커서는 왼쪽, 오른쪽 순서로 각각 약 1.5 s 머뭅니다.
//   - 호버 시 중심 기준 배율 1→1.15를 0.5 s 동안 적용합니다(cubic ease-out). 벗어나면 0.5 s에 복귀합니다.
//   - 틀 밖의 내용은 클리핑합니다(overflow: hidden). 틀과 캡션은 움직이지 않습니다.
// 활용 상황: 작업물·제품·기사 사진 목록에서 선택하려는 항목을 표시할 때 사용합니다.
// 태그: hover, zoom, image, scale, overflow-hidden, thumbnail, gallery
// 크기: 539x208 px · 반복 주기: 4.8 s
// 필요한 코드: kit lp -> https://comnori.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#082
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 10
// 10 ホバーで画像ズーム（枠はそのままで、中の写真だけがゆっくり拡大する）
CELLS[9] = (k, w, h, u) => {
  frame(k);
  const FX = [22, 267], FY = 16, FW = 222, FH = 118, ZOOM = 0.15;
  const cur = t => path([[0, 60, 160], [0.5, 120, 70], [1.8, 140, 84], [2.2, 370, 70], [3.4, 390, 84], [3.9, 470, 160], [4.8, 60, 160]], t);
  view(k, () => {
    for (let i = 0; i < 2; i++) {
      const inside = t => { const [x, y] = cur(t); return x > FX[i] && x < FX[i] + FW && y > FY && y < FY + FH; };
      const q = trans(inside, u, 0.5);
      pic(k, FX[i], FY, FW, FH, i, 8, 1 + ZOOM * q);
      txt(k, ['夏の山小屋', '海辺のカフェ'][i], FX[i], FY + FH + 20, `800 13px ${JP}`, C.ink); bars(k, FX[i], FY + FH + 28, 150, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
