// 동작 사양:
//   - 542×191 카드에서 4.5 s 주기. (24, 112)의 파란 윤곽선 “상세” 버튼(108×36)을 1.0 s에 클릭합니다.
//   - 표시 정도 a: 1.0–1.28 s에 easeOutCubic으로 0→1. 불투명도 a, 오른쪽에서 6 px 이동해 들어옵니다.
//   - 패널: 흰색 300×120, 모서리 반지름 10, 위치 (156, 62), 테두리 1 px, 그림자 블러 22, y 오프셋 6. 왼쪽 경계(y 122–138)의 너비 10 삼각형이 버튼을 가리킵니다.
//   - 내부: 굵은 제목 막대, 글 두 줄, 윤곽선 버튼 70×26, 파란 OK 버튼 72×26.
//   - 닫기: 커서가 3.0 s까지 OK (402, 157)로 이동하고 3.3 s에 클릭합니다. a에 (1 − 3.3–3.5 s의 easeInOutCubic)을 곱합니다.
// 활용 상황: 짧은 설정이나 확인을 관련 버튼 바로 옆에서 처리하게 할 때 사용합니다.
// 태그: popover, popup, callout, bubble, flyout, overlay
// 크기: 542x191 px · 반복 주기: 4.5 s
// 필요한 코드: kit ui -> https://comnori.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#055
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 07
// 07 ポップオーバー（Popover）
CELLS[6] = ui((w, h, t) => {
  pageLines(24, 16, 494);
  strokeRR(24, 112, 108, 36, 8, C.blue, 2); text('詳細', 78, 130, 15, C.blue, 600, 'center');
  fillRR(24, 160, 494, 22, 6, C.block);
  const a = eo(seg(t, 1.0, 1.28)) * (1 - eio(seg(t, 3.3, 3.5)));
  if (a > 0) {
    X.save(); X.globalAlpha = a; X.translate(6 * (1 - a), 0);
    const px = 156, py = 62, pw = 300, ph = 120;
    shadowed(() => { fillRR(px, py, pw, ph, 10, '#fff'); }, 22, 6, .2); strokeRR(px, py, pw, ph, 10, C.border, 1);
    X.beginPath(); X.moveTo(px + 1, 122); X.lineTo(px - 10, 130); X.lineTo(px + 1, 138); X.fillStyle = '#fff'; X.fill();
    X.beginPath(); X.moveTo(px, 122); X.lineTo(px - 10, 130); X.lineTo(px, 138); X.strokeStyle = C.border; X.lineWidth = 1; X.stroke();
    bar(px + 18, py + 18, 130, C.ink, 11); bar(px + 18, py + 42, 250); bar(px + 18, py + 60, 200);
    strokeRR(px + 130, py + 82, 70, 26, 6, C.line2, 1.5); fillRR(px + 210, py + 82, 72, 26, 6, C.blue); text('OK', px + 246, py + 95, 14, '#fff', 700, 'center', NUMF);
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 84, 134], [1.6, 84, 134], [3.0, 402, 157], [3.5, 402, 157], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
