// 교차 편집 / Cross-cutting
// 팩: camera2 (카메라 움직임과 장면 전환 2) · element 22 · 분류: 전환과 편집
// 동작 요약: 달리는 사람과 다가오는 자동차를 점점 짧은 간격으로 번갈아 보여 주어 긴장감을 높입니다.
// 동작 사양:
//   - 두 장면은 같은 시계를 공유합니다. A의 달리는 사람은 왼쪽 0.3 m에서 깊이 15 → 4.5 m로 이동하며 팔다리는 u×12로 움직입니다. B의 자동차는 도시의 오른쪽 1 m에서 깊이 17 → 4.5 m로 이동합니다. 두 이동 모두 4.7 s 동안 진행됩니다.
//   - 컷 시점은 0 / 1.1 / 2.0 / 2.7 / 3.3 / 3.8 / 4.2 / 4.5 s입니다. 간격은 1.1 → 0.9 → 0.7 → 0.6 → 0.5 → 0.4 → 0.3 s로 짧아집니다.
//   - 화면에 보이지 않는 동안에도 각 장면의 동작은 계속 진행됩니다.
//   - A는 높이 1.3 m의 고정 카메라와 기둥 네 개, B는 높이 1.6 m·피치 +0.06 rad의 고정 카메라와 도시·태양으로 구성합니다.
//   - 하단의 여덟 구간은 회색 A와 파란 B가 번갈아 나타나며 재생 헤드가 이동합니다. 마지막 0.3 s 동안 A로 페이드합니다.
//   - 상단에는 ‘SCENE A / B’를 표시합니다.
// 활용 상황: 추격·구조처럼 동시에 진행되는 사건을 연결하고 긴장감을 높일 때 사용합니다.
// 태그: cross-cutting, parallel editing, intercut, parallel action, chase, edit
// 크기: 539x208 px · 반복 주기: 9.4 s
// 필요한 코드: kit camera2 -> https://comnori.github.io/motion-prompt/api/v1/kit/camera2.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#190
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 22
// 22 クロスカッティング: 同時に進む2つの場面（走る人物・走る車）を、だんだん短く交互に見せる
const XC = [0, 1.1, 2.0, 2.7, 3.3, 3.8, 4.2, 4.5, CYC];
CELLS[21] = shot((k, w, h, u) => {
  const r = P(u, CYC - 0.3, CYC);
  const runA = (kk, t) => world(kk, w, h, cam({ y: 1.3 }), { person: { x: -0.3, z: lerp(15, 4.5, t / CYC), walk: t * 12 }, fg: [[-2.2, 3.5], [-2.2, 6.5], [-2.2, 9.5], [-2.2, 12.5]] });
  const carB = (kk, t) => world(kk, w, h, cam({ y: 1.6, pitch: 0.06 }), { city: true, mountains: false, trees: false, sun: [-26, 40, 150], extra: [[1.0, 0, lerp(17, 4.5, t / CYC), CAR[3]]] });
  let i = 0; while (u >= XC[i + 1]) i++;
  const isA = i % 2 === 0;
  blend(k, kk => (isA ? runA : carB)(kk, u), kk => runA(kk, 0), eIOs(r));   // 最後の0.3秒で A の開始状態へ戻す
  const segs = []; for (let j = 0; j < XC.length - 1; j++) segs.push([XC[j], XC[j + 1], j % 2 ? 'B' : 'A', j % 2 ? C.blue : C.g2]);
  strip(k, w, h, segs, u);
  return r > .5 || isA ? 'SCENE A' : 'SCENE B';
});
