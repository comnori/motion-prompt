// 분할 화면 / Split screen
// 팩: motion2 (모션 그래픽 움직임 2) · element 24 · 분류: 전환과 강조
// 동작 요약: 전체 화면의 한 장면이 두 패널, 세 패널로 나뉘고 각 패널 안에서 다른 장면이 움직입니다.
// 동작 사양:
//   - 전체 바다 장면을 0.45–0.95 s에 두 패널로 나누고(구분선 539→269.5), 1.4–1.9 s에 세 패널로 나눕니다(구분선 1/3·2/3). 3.75–4.4 s에 다시 하나로 합칩니다. 모두 cubic ease-in-out입니다.
//   - 바다·도시·들판은 각 패널 중앙에 맞춰 클리핑하고 계속 움직입니다(파도와 돛단배, 4.7 s마다 8번 바뀌는 도시 창문, 4.7 s마다 두 바퀴 도는 풍차).
//   - 패널 사이에 두께 6의 흰 구분선을 둡니다.
// 활용 상황: 변경 전후·선택지 비교, 동시 상황, 여러 사람이나 장소를 나란히 보여 줄 때 사용합니다.
// 태그: split-screen, multi-screen, panels, side-by-side, comparison, layout
// 크기: 539x208 px · 반복 주기: 4.7 s
// 필요한 코드: kit motion2 -> https://comnori.github.io/motion-prompt/api/v1/kit/motion2.js
// 움직이는 예제: https://comnori.github.io/motion-prompt/?lang=ko#144
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (호출하는 쪽에서 화면 지우기와 save/restore 처리), w·h = 크기, u = 초 (0 이상, 반복 주기 미만). u만으로 화면이 결정됩니다.
// 아래부터 항목 코드입니다. CELLS, 상수와 보조 함수는 kit에 있습니다.

// @element 24
// 24 スプリットスクリーン（1枚の画面が2つ、3つに分かれて、別々の場面が同時に動く）
CELLS[23] = (k, w, h, u) => {
  u = snapU(u);
  const s2 = eIO(P(u, 0.45, 0.95)), s3 = eIO(P(u, 1.4, 1.9)), m = eIO(P(u, 3.75, 4.4));
  let b1 = lerp(lerp(w, w / 2, s2), w / 3, s3), b2 = lerp(w, 2 * w / 3, s3);
  b1 = lerp(b1, w, m); b2 = lerp(b2, w, m);
  pnSea(k, 0, b1, h, u, false);
  if (b1 < w - 0.5) pnCity(k, b1, Math.max(b1, b2), h, u, false);
  if (b2 < w - 0.5) pnField(k, b2, w, h, u, false);
  k.fillStyle = '#fff';[b1, b2].forEach(b => { if (b < w - 0.5) k.fillRect(b - 3, 0, 6, h); });
};
